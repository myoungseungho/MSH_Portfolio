야생동물이 서로 잡아먹어요

생태계 시뮬레이션 (Wildlife Ecosystem)

Q: 동물이 그냥 랜덤하게 돌아다니는 게 아니라고요?
전혀 아닙니다. RDR2의 야생동물은 실제 생태계처럼 작동합니다. 늑대는 사슴을 사냥하고, 독수리는 시체를 먹고, 곰은 영역을 지킵니다. 플레이어가 사슴을 너무 많이 사냥하면 그 지역에서 사슴이 사라지고, 늑대도 먹이를 찾아 이동합니다.

Part 1: 200종 이상의 동물

Q: 동물이 몇 종류나 있나요?
RDR2의 동물 다양성:
  • 포유류: 곰, 늑대, 사슴, 엘크, 들소, 쿠거, 너구리, 토끼, 비버 등 60+종
  • 조류: 독수리, 매, 까마귀, 오리, 거위, 칠면조, 앵무새 등 80+종
  • 파충류: 악어, 뱀, 거북이, 도마뱀 등 15+종
  • 양서류: 개구리, 두꺼비 등 5+종
  • 어류: 농어, 연어, 메기, 송어 등 30+종
  • 무척추동물: 게, 가재 등 10+종
총 200종 이상, 각각 고유한 행동 패턴과 생태적 역할을 가집니다.
생태계 계층 구조 ┌─────────────────────────────────────┐ │ 최상위 포식자 (Apex) │ │ 곰, 쿠거, 늑대, 악어 │ └────────────┬────────────────────────┘ │ (사냥) ▼ ┌─────────────────────────────────────┐ │ 초식 동물 (Herbivore) │ │ 사슴, 엘크, 들소, 양, 말 │ └────────────┬────────────────────────┘ │ (먹음) ▼ ┌─────────────────────────────────────┐ │ 식물 (Flora) │ │ 풀, 관목, 열매 │ └─────────────────────────────────────┘ ┌─────────────────────────────────────┐ │ 청소 동물 (Scavenger) │ │ 독수리, 까마귀, 여우, 코요테 │ │ (시체를 먹음) │ └─────────────────────────────────────┘ 특이 케이스: - 악어: 물가 최상위 포식자 - 독수리: 하늘 포식자 + 청소부 - 늑대: 무리 사냥 (협동) - 곰: 잡식 (물고기 + 열매 + 사냥)

Part 2: 먹이사슬 시스템

Q: 동물끼리 실제로 사냥하나요?
네! 플레이어가 숲을 걷다가 늑대가 사슴을 물어뜯는 장면, 독수리가 하늘에서 급강하해 토끼를 채가는 장면, 악어가 물가의 사슴을 물속으로 끌고 가는 장면을 목격할 수 있습니다. 이 모든 것이 스크립트 없이 자동으로 발생합니다. 
// 의사코드: 포식자 사냥 AI
class PredatorAI {
    Animal owner;
    float hunger = 50.0f;       // 0 (배부름) ~ 100 (굶주림)
    float hungerThreshold = 60.0f;

    void Update(float deltaTime) {
        // 1. 배고픔 증가 (시간당 10)
        hunger += deltaTime * (10.0f / 3600.0f);

        // 2. 배고프면 사냥 모드
        if (hunger > hungerThreshold) {
            EnterHuntMode();
        } else {
            EnterIdleMode();  // 휴식, 순찰
        }
    }

    void EnterHuntMode() {
        // 1. 먹이 검색 (시야 범위 내)
        Animal[] nearbyAnimals = FindAnimalsInRadius(owner.position, 50.0f);

        Animal bestPrey = null;
        float bestScore = 0.0f;

        for (Animal animal : nearbyAnimals) {
            // 먹이 후보인지 확인
            if (!IsPrey(animal.species)) continue;

            // 점수 계산 (거리, 건강, 크기)
            float score = EvaluatePreyScore(animal);
            if (score > bestScore) {
                bestScore = score;
                bestPrey = animal;
            }
        }

        // 2. 먹이 발견 → 추적
        if (bestPrey != null) {
            owner.SetState(State.Chasing);
            owner.targetPrey = bestPrey;
            ChaseAndAttack(bestPrey);
        }
        // 3. 먹이 없음 → 순찰
        else {
            Patrol();
        }
    }

    float EvaluatePreyScore(Animal prey) {
        float score = 100.0f;

        // 거리 (가까울수록 좋음)
        float distance = Distance(owner.position, prey.position);
        score -= distance * 0.5f;

        // 건강 (약할수록 좋음)
        score += (100 - prey.health) * 0.3f;

        // 크기 (적당한 크기가 좋음)
        if (prey.size > owner.size * 2) {
            score -= 50.0f;  // 너무 크면 위험
        }

        // 무리 (혼자 있는 개체 선호)
        if (prey.IsInHerd()) {
            score -= 30.0f;
        }

        return score;
    }

    void ChaseAndAttack(Animal prey) {
        // 1. 추적
        Vector3 direction = (prey.position - owner.position).Normalized();
        owner.MoveTowards(prey.position, owner.runSpeed);

        // 2. 사정거리 내 → 공격
        if (Distance(owner.position, prey.position) < owner.attackRange) {
            Attack(prey);
        }
    }

    void Attack(Animal prey) {
        // 공격 애니메이션
        owner.PlayAnimation("Attack_Bite");

        // 데미지
        float damage = owner.attackPower * Random.Range(0.8f, 1.2f);
        prey.TakeDamage(damage);

        // 사냥 성공 → 먹이 먹기
        if (prey.health <= 0) {
            OnKillSuccess(prey);
        }
    }

    void OnKillSuccess(Animal prey) {
        // 1. 먹이 먹기
        owner.PlayAnimation("Eating");
        hunger -= 50.0f;  // 배고픔 감소

        // 2. 시체 생성 (다른 동물이 먹을 수 있음)
        Carcass carcass = SpawnCarcass(prey.species, prey.position);
        carcass.remainingMeat = prey.size * 5.0f;

        // 3. 포만감 → 휴식 모드
        owner.SetState(State.Resting);
    }

    bool IsPrey(AnimalSpecies species) {
        // 늑대의 먹이 목록
        if (owner.species == AnimalSpecies.Wolf) {
            return species == AnimalSpecies.Deer ||
                   species == AnimalSpecies.Rabbit ||
                   species == AnimalSpecies.Sheep;
        }

        // 쿠거의 먹이 목록
        if (owner.species == AnimalSpecies.Cougar) {
            return species == AnimalSpecies.Deer ||
                   species == AnimalSpecies.Boar;
        }

        // ... 각 포식자마다 먹이 목록 정의
        return false;
    }
}

Part 3: 영역 시스템 (Territory)

Q: 동물이 맵 전체를 돌아다니나요?
아니요. 각 동물은 서식지 (Territory)를 가집니다. 곰은 숲 깊은 곳, 악어는 늪지대, 들소는 평원에만 나타납니다. 서식지를 벗어나면 자동으로 되돌아옵니다.
영역 시스템 예시: Grizzlies 산맥 ┌────────────────────────────────────┐ │ Grizzlies Mountains │ │ │ │ 🐻 (곰 영역) │ │ ├─ 중심: (1200, 850) │ │ ├─ 반경: 500m │ │ ├─ 개체수: 3~5마리 │ │ └─ 행동: 순찰, 낚시, 영역 방어 │ │ │ │ 🦌 (사슴 영역) │ │ ├─ 중심: (1100, 900) │ │ ├─ 반경: 800m │ │ ├─ 개체수: 15~25마리 │ │ └─ 행동: 풀 먹기, 무리 이동 │ │ │ │ 🐺 (늑대 영역) │ │ ├─ 중심: (1150, 880) │ │ ├─ 반경: 1000m │ │ ├─ 개체수: 4~8마리 (무리) │ │ └─ 행동: 사냥, 협동 │ └────────────────────────────────────┘ 영역 겹침: - 사슴과 늑대 영역 겹침 → 포식 관계 - 곰과 늑대 영역 겹침 → 경쟁 (같은 먹이) - 곰은 영역에서 강함 → 늑대 쫓아냄 
// 의사코드: 영역 시스템
class Territory {
    Vector3 center;
    float radius;
    AnimalSpecies species;
    int minPopulation;
    int maxPopulation;
    List residents;

    bool IsInTerritory(Vector3 position) {
        return Distance(position, center) <= radius;
    }

    void MaintainPopulation() {
        // 1. 개체수 확인
        int aliveCount = residents.Count(a => a.IsAlive());

        // 2. 개체수 부족 → 스폰
        if (aliveCount < minPopulation) {
            int toSpawn = minPopulation - aliveCount;
            for (int i = 0; i < toSpawn; i++) {
                SpawnAnimal();
            }
        }

        // 3. 개체수 초과 → 이주 (다른 영역으로)
        if (aliveCount > maxPopulation) {
            Animal weakest = residents.OrderBy(a => a.health).First();
            weakest.Migrate(FindNearbyTerritory());
        }
    }

    Vector3 GetRandomSpawnPoint() {
        // 영역 내 랜덤 위치
        Vector3 randomOffset = Random.InsideUnitSphere() * radius;
        return center + randomOffset;
    }
}

class Animal {
    Territory homeTerritory;
    Vector3 position;

    void Update(float deltaTime) {
        // ... 행동 로직 ...

        // 영역 이탈 확인
        if (!homeTerritory.IsInTerritory(position)) {
            ReturnToTerritory();
        }
    }

    void ReturnToTerritory() {
        // 영역 중심으로 이동
        Vector3 direction = (homeTerritory.center - position).Normalized();
        MoveTowards(homeTerritory.center, runSpeed);
    }
}

// 영역 초기화 예시
void InitializeGrizzliesTerritories() {
    // 곰 영역
    Territory bearTerritory = new Territory {
        center = new Vector3(1200, 850),
        radius = 500.0f,
        species = AnimalSpecies.GrizzlyBear,
        minPopulation = 3,
        maxPopulation = 5
    };

    // 사슴 영역
    Territory deerTerritory = new Territory {
        center = new Vector3(1100, 900),
        radius = 800.0f,
        species = AnimalSpecies.Deer,
        minPopulation = 15,
        maxPopulation = 25
    };

    // 늑대 영역
    Territory wolfTerritory = new Territory {
        center = new Vector3(1150, 880),
        radius = 1000.0f,
        species = AnimalSpecies.Wolf,
        minPopulation = 4,
        maxPopulation = 8
    };
}

Part 4: 시간/날씨 반응

Q: 동물이 밤낮을 구분하나요?
네! 동물마다 일주기 (Circadian Rhythm)가 있습니다. 야행성 동물은 밤에 활발하고, 주행성 동물은 낮에 활발합니다. 날씨에도 반응해서 비가 오면 숨고, 눈이 오면 털이 두꺼워집니다.
동물별 시간 패턴:
  • 사슴 (주행성): 아침/저녁 활발 (06:00-09:00, 17:00-20:00)
  • 늑대 (야행성): 밤에 사냥 (20:00-04:00)
  • 곰 (낮 활동): 낮에 낚시, 밤에 휴식
  • 박쥐 (야행성): 일몰 후 등장 (19:00-05:00)
  • 독수리 (주행성): 낮에 하늘 순찰 (08:00-18:00)
// 의사코드: 시간/날씨 반응
class AnimalCircadianRhythm {
    Animal owner;
    TimeOfDay currentTime;
    Weather currentWeather;

    void Update() {
        currentTime = GameWorld.GetTimeOfDay();
        currentWeather = GameWorld.GetWeather();

        // 1. 시간대별 활동성
        float activityLevel = GetActivityLevel(currentTime);

        // 2. 날씨 영향
        activityLevel *= GetWeatherModifier(currentWeather);

        // 3. 행동 조절
        owner.movementSpeed *= activityLevel;
        owner.alertness *= activityLevel;

        // 4. 특정 행동 트리거
        if (ShouldSleep(currentTime, activityLevel)) {
            owner.EnterSleepMode();
        }
        if (ShouldSeekShelter(currentWeather)) {
            owner.FindShelter();
        }
    }

    float GetActivityLevel(TimeOfDay time) {
        if (owner.species == AnimalSpecies.Deer) {
            // 사슴: 새벽/저녁에 활발
            if (time.hour >= 6 && time.hour <= 9) return 1.0f;   // 아침
            if (time.hour >= 17 && time.hour <= 20) return 1.0f; // 저녁
            if (time.hour >= 12 && time.hour <= 16) return 0.3f; // 한낮
            return 0.5f;  // 밤
        }

        if (owner.species == AnimalSpecies.Wolf) {
            // 늑대: 밤에 활발
            if (time.hour >= 20 || time.hour <= 4) return 1.0f;  // 밤
            if (time.hour >= 6 && time.hour <= 18) return 0.4f;  // 낮
            return 0.7f;  // 황혼
        }

        if (owner.species == AnimalSpecies.GrizzlyBear) {
            // 곰: 낮에 활동
            if (time.hour >= 8 && time.hour <= 18) return 1.0f;  // 낮
            return 0.2f;  // 밤 (잠)
        }

        return 1.0f;  // 기본
    }

    float GetWeatherModifier(Weather weather) {
        if (weather == Weather.Rain) {
            // 비: 활동성 감소
            return 0.5f;
        }

        if (weather == Weather.Snow) {
            // 눈: 종에 따라 다름
            if (owner.species == AnimalSpecies.Wolf) {
                return 1.2f;  // 늑대는 눈에서 더 활발
            }
            if (owner.species == AnimalSpecies.Alligator) {
                return 0.1f;  // 악어는 추위에 약함
            }
            return 0.7f;
        }

        if (weather == Weather.Fog) {
            // 안개: 시야 제한 → 경계
            owner.alertness *= 1.5f;
            return 0.8f;
        }

        return 1.0f;  // 맑음
    }

    bool ShouldSeekShelter(Weather weather) {
        // 비/눈 → 동굴, 나무 아래 등
        if (weather == Weather.Rain || weather == Weather.Snow) {
            return Random.value < 0.7f;  // 70% 확률
        }
        return false;
    }

    void FindShelter() {
        // 1. 근처 은신처 검색
        Shelter[] shelters = FindSheltersInRadius(owner.position, 100.0f);

        if (shelters.Length > 0) {
            // 가장 가까운 곳으로
            Shelter nearest = shelters.OrderBy(s => Distance(owner.position, s.position)).First();
            owner.MoveTowards(nearest.position, owner.walkSpeed);
        }
        // 2. 은신처 없으면 나무 아래
        else {
            Tree nearestTree = FindNearestTree(owner.position);
            if (nearestTree != null) {
                owner.MoveTowards(nearestTree.position, owner.walkSpeed);
            }
        }
    }
}

Part 5: 플레이어 영향 (생태계 균형)

Q: 플레이어가 사냥을 많이 하면 어떻게 되나요?
개체수가 실제로 감소합니다. 특정 지역에서 사슴을 계속 사냥하면:
  1. 사슴 개체수 감소 (20마리 → 5마리)
  2. 늑대가 먹이 부족 → 다른 지역으로 이동
  3. 사슴 없는 풀밭 → 식생 회복 (풀이 더 자람)
  4. 일정 기간 후 사슴 개체수 서서히 회복 (번식)
이것은 Lotka-Volterra 모델 (포식자-피식자 방정식)을 단순화한 것입니다.
생태계 균형 시뮬레이션 시나리오: 플레이어가 사슴 15마리 사냥 Day 1 사슴: ■■■■■■■■■■■■■■■ (15마리) 늑대: ■■■■ (4마리) Day 3 (사냥 후) 사슴: ■■■■■ (5마리) ← 급감! 늑대: ■■■■ (4마리) ← 아직 유지 Day 7 사슴: ■■■■ (4마리) ← 계속 감소 (늑대 사냥) 늑대: ■■■ (3마리) ← 먹이 부족 → 1마리 이주 Day 14 사슴: ■■■■■■ (6마리) ← 번식 시작 늑대: ■■ (2마리) ← 계속 이주 Day 30 사슴: ■■■■■■■■■■ (10마리) ← 회복 중 늑대: ■■■ (3마리) ← 사슴 증가 → 돌아옴 Day 60 사슴: ■■■■■■■■■■■■■■ (14마리) ← 거의 회복 늑대: ■■■■ (4마리) ← 원래대로 
// 의사코드: 생태계 균형 시뮬레이션
class EcosystemSimulator {
    Territory territory;
    float preyPopulation;
    float predatorPopulation;

    void Update(float deltaTime) {
        // Lotka-Volterra 방정식 단순화
        float dt = deltaTime / 86400.0f;  // 일 단위

        // 1. 피식자 변화율
        //    dPrey/dt = a * Prey - b * Prey * Predator - c * PlayerHunting
        float preyBirthRate = 0.1f;       // a: 번식률 (10%/일)
        float predationRate = 0.02f;      // b: 포식률
        float playerHunting = GetPlayerHuntingRate();

        float preyChange = (preyBirthRate * preyPopulation)
                         - (predationRate * preyPopulation * predatorPopulation)
                         - playerHunting;

        preyPopulation += preyChange * dt;

        // 2. 포식자 변화율
        //    dPredator/dt = d * Prey * Predator - e * Predator
        float predatorEfficiency = 0.01f;  // d: 사냥 효율
        float predatorDeathRate = 0.05f;   // e: 자연 사망률

        float predatorChange = (predatorEfficiency * preyPopulation * predatorPopulation)
                             - (predatorDeathRate * predatorPopulation);

        predatorPopulation += predatorChange * dt;

        // 3. 개체수 제한
        preyPopulation = Clamp(preyPopulation, 0, territory.maxPopulation);
        predatorPopulation = Clamp(predatorPopulation, 0, territory.maxPopulation * 0.3f);

        // 4. 실제 동물 스폰/제거
        SyncAnimalCount();
    }

    float GetPlayerHuntingRate() {
        // 최근 7일간 플레이어 사냥 통계
        int hunted = GameStats.GetHuntedCount(territory, days: 7);
        return hunted / 7.0f;  // 일평균
    }

    void SyncAnimalCount() {
        // 피식자 (사슴)
        int currentPrey = territory.GetAnimalCount(AnimalSpecies.Deer);
        int targetPrey = (int)preyPopulation;

        if (currentPrey < targetPrey) {
            // 스폰
            for (int i = 0; i < targetPrey - currentPrey; i++) {
                territory.SpawnAnimal(AnimalSpecies.Deer);
            }
        } else if (currentPrey > targetPrey) {
            // 제거 (자연사 시뮬레이션)
            Animal weakest = territory.GetWeakestAnimal(AnimalSpecies.Deer);
            weakest.Die(DeathCause.Natural);
        }

        // 포식자 (늑대)
        int currentPredator = territory.GetAnimalCount(AnimalSpecies.Wolf);
        int targetPredator = (int)predatorPopulation;

        if (currentPredator < targetPredator) {
            territory.SpawnAnimal(AnimalSpecies.Wolf);
        } else if (currentPredator > targetPredator) {
            // 이주
            Animal wolf = territory.GetRandomAnimal(AnimalSpecies.Wolf);
            wolf.Migrate(FindNearbyTerritory());
        }
    }
}

Part 6: 성능 최적화 (Grid 기반)

Q: 수천 마리 동물을 어떻게 시뮬레이션하나요?
맵 전체를 Grid (그리드)로 나누고, 각 셀마다 동물을 관리합니다. 플레이어 근처 셀만 풀로 시뮬레이션하고, 먼 셀은 통계만 업데이트합니다.
Grid 기반 최적화 맵 분할 (100m × 100m 셀) ┌──────┬──────┬──────┬──────┐ │ A1 │ A2 │ A3 │ A4 │ 플레이어: B2 │ 🦌 │ 🐺 │ │ │ ├──────┼──────┼──────┼──────┤ │ B1 │ B2🧍 │ B3 │ B4 │ Active 셀: B1, B2, B3 │ 🐻 │ 🦌🦌 │ 🐺 │ │ (플레이어 인접) ├──────┼──────┼──────┼──────┤ │ C1 │ C2 │ C3 │ C4 │ Simulated 셀: A1~C4 │ │ 🦌 │ │ 🐻 │ (멀리 있음) ├──────┼──────┼──────┼──────┤ │ D1 │ D2 │ D3 │ D4 │ Paused 셀: D1~D4 │ │ │ │ │ (매우 멀리) └──────┴──────┴──────┴──────┘ 처리 방식: - Active (0-200m): 풀 AI + 물리 + 렌더링 - Simulated (200-500m): 통계만 (개체수 유지) - Paused (500m+): 업데이트 안 함 
// 의사코드: Grid 기반 동물 관리
class WildlifeGrid {
    struct GridCell {
        Vector2Int coordinates;
        List animals;
        EcosystemSimulator ecosystem;
        float lastUpdateTime;
    };

    GridCell[,] grid;
    float cellSize = 100.0f;

    void Update(float deltaTime) {
        Vector3 playerPos = Player.Instance.position;
        Vector2Int playerCell = WorldToGrid(playerPos);

        for (int x = 0; x < gridWidth; x++) {
            for (int y = 0; y < gridHeight; y++) {
                GridCell cell = grid[x, y];

                // 플레이어로부터 거리 계산
                float distance = Distance(GridToWorld(x, y), playerPos);

                // LOD 0: Active (0-200m)
                if (distance < 200.0f) {
                    UpdateActive(cell, deltaTime);
                }

                // LOD 1: Simulated (200-500m)
                else if (distance < 500.0f) {
                    // 1초당 1회만 업데이트
                    if (Time.time - cell.lastUpdateTime > 1.0f) {
                        UpdateSimulated(cell, deltaTime);
                        cell.lastUpdateTime = Time.time;
                    }
                }

                // LOD 2: Paused (500m+)
                // → 업데이트 안 함
            }
        }
    }

    void UpdateActive(GridCell cell, float deltaTime) {
        // 모든 동물 Full AI
        for (Animal animal : cell.animals) {
            animal.UpdateAI(deltaTime);      // Behavior Tree
            animal.UpdatePhysics(deltaTime);  // 이동, 충돌
            animal.UpdateAnimation(deltaTime);
        }

        // 생태계 시뮬레이션
        cell.ecosystem.Update(deltaTime);
    }

    void UpdateSimulated(GridCell cell, float deltaTime) {
        // 생태계 통계만 업데이트
        cell.ecosystem.Update(deltaTime);

        // 동물 개체 업데이트 안 함 (멈춤)
        // 플레이어가 가까이 오면 "재개"
    }

    Vector2Int WorldToGrid(Vector3 worldPos) {
        return new Vector2Int(
            (int)(worldPos.x / cellSize),
            (int)(worldPos.z / cellSize)
        );
    }

    Vector3 GridToWorld(int x, int y) {
        return new Vector3(
            x * cellSize + cellSize * 0.5f,
            0,
            y * cellSize + cellSize * 0.5f
        );
    }
}

핵심 요약

  1. 먹이사슬: 200종 동물이 포식자-피식자 관계로 연결
  2. 영역 시스템: 각 종은 고유 서식지에서만 활동 (Territory)
  3. 시간/날씨: 야행성/주행성 구분, 날씨에 따라 행동 변화
  4. 생태계 균형: 플레이어 사냥 → 개체수 변화 → Lotka-Volterra 모델
  5. Grid 최적화: 거리별로 Full AI / 통계 / 정지 모드 전환

게임 내 체험: Tall Trees 숲에서 사슴 무리를 발견하면, 가만히 지켜보세요. 얼마 지나지 않아 늑대 무리가 나타나 사냥을 시작할 수 있습니다. 사슴은 무리를 지어 도망가고, 늑대는 협동해서 약한 개체를 고립시킵니다. 사냥 성공 후 늑대가 먹다 남긴 시체에는 독수리와 까마귀가 모여듭니다.

기술적 의의: RDR2의 생태계는 단순한 "랜덤 스폰"이 아니라, 실제 생물학적 원리(포식, 번식, 영역)를 게임에 구현한 사례입니다. 플레이어는 세계의 "관찰자"이자 "참여자"로서, 생태계 균형에 영향을 줍니다.
