Part 2. RAM vs VRAM - 두 개의 메모리 세계

RAM과 VRAM이 따로 있다는 건 알겠는데, 정확히 뭐가 다른 거야?

게임 개발에서 RAM (System Memory)과 VRAM (Video Memory)은 완전히 다른 용도로 사용됩니다:

RAM과 VRAM 사이에 데이터 이동은 어떻게 일어나?

에셋 로딩 시 데이터는 디스크 → RAM → VRAM 순서로 이동합니다. 이 과정에서 병목이 발생할 수 있습니다:

메모리 예산은 코드로 어떻게 관리해?

각 시스템이 자체 예산 내에서만 할당하도록 Budget Manager를 구현합니다:

// 메모리 예산 관리 시스템 enum class MemoryCategory { WorldStreaming, TextureStreaming, AISystem, PhysicsSimulation, AudioSystem, GameLogic, Network, UI, Emergency, Count }; class MemoryBudgetManager { private: struct CategoryBudget { size_t maxBytes; std::atomic<size_t> usedBytes{0}; float warningThreshold = 0.85f; float criticalThreshold = 0.95f; }; std::array<CategoryBudget, (size_t)MemoryCategory::Count> m_budgets; std::function<void(MemoryCategory)> m_onBudgetExceeded; public: void Initialize(size_t totalAvailable) { // 예산 배분 m_budgets[(size_t)MemoryCategory::WorldStreaming].maxBytes = totalAvailable * 0.29; m_budgets[(size_t)MemoryCategory::TextureStreaming].maxBytes = totalAvailable * 0.14; m_budgets[(size_t)MemoryCategory::AISystem].maxBytes = totalAvailable * 0.11; // ... 나머지 카테고리 } // 할당 요청 (예산 초과 시 실패 가능) bool TryAllocate(MemoryCategory category, size_t bytes) { auto& budget = m_budgets[(size_t)category]; size_t current = budget.usedBytes.load(); size_t newUsed = current + bytes; if (newUsed > budget.maxBytes) { // 예산 초과! if (m_onBudgetExceeded) { m_onBudgetExceeded(category); } return false; } // CAS로 원자적 업데이트 while (!budget.usedBytes.compare_exchange_weak(current, newUsed)) { newUsed = current + bytes; if (newUsed > budget.maxBytes) { return false; } } // 경고 임계값 체크 float usage = (float)newUsed / budget.maxBytes; if (usage >= budget.criticalThreshold) { LogCritical("Memory CRITICAL: %s at %.1f%%", GetCategoryName(category), usage * 100); } else if (usage >= budget.warningThreshold) { LogWarning("Memory WARNING: %s at %.1f%%", GetCategoryName(category), usage * 100); } return true; } void Free(MemoryCategory category, size_t bytes) { m_budgets[(size_t)category].usedBytes -= bytes; } // 전체 상태 보고 MemoryReport GenerateReport() { MemoryReport report; for (size_t i = 0; i < (size_t)MemoryCategory::Count; i++) { report.categories[i] = { .name = GetCategoryName((MemoryCategory)i), .used = m_budgets[i].usedBytes.load(), .max = m_budgets[i].maxBytes, .percentage = (float)m_budgets[i].usedBytes / m_budgets[i].maxBytes }; } return report; } };

Part 4. 메모리 압박 (Memory Pressure) 대응

메모리가 부족해지면 어떻게 대응해?

메모리 압박 상황에서는 단계적 대응이 필요합니다. 너무 급격한 대응은 스터터링을 유발하고, 너무 늦은 대응은 크래시를 유발합니다:

메모리 압박 단계별 대응:

┌─────────────────────────────────────────────────────────────┐
│ 단계 0: 정상 (0% ~ 70%)                                     │
├─────────────────────────────────────────────────────────────┤
│ 상태: 여유로움                                              │
│ 동작: 일반 스트리밍, 최고 품질 텍스처                       │
│ 캐시: 적극적 프리로딩                                       │
│                                                             │
│ [███████████████████████░░░░░░░░░░░░░░] 65%                 │
└─────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────┐
│ 단계 1: 주의 (70% ~ 85%)                                    │
├─────────────────────────────────────────────────────────────┤
│ 상태: 주의 필요                                             │

// 메모리 추적 시스템 class MemoryTracker { private: struct AllocationRecord { void* address; size_t size; const char* file; int line; const char* tag; uint64_t timestamp; uint32_t frameNumber; }; std::unordered_map<void*, AllocationRecord> m_allocations; std::unordered_map<std::string, size_t> m_taggedUsage; std::mutex m_mutex; public: void* TrackAlloc(size_t size, const char* file, int line, const char* tag) { void* ptr = malloc(size); if (ptr) { std::lock_guard<std::mutex> lock(m_mutex); m_allocations[ptr] = { .address = ptr, .size = size, .file = file, .line = line, .tag = tag, .timestamp = GetCurrentTimestamp(), .frameNumber = g_FrameNumber }; m_taggedUsage[tag] += size; } return ptr; } void TrackFree(void* ptr) { std::lock_guard<std::mutex> lock(m_mutex); auto it = m_allocations.find(ptr); if (it != m_allocations.end()) { m_taggedUsage[it->second.tag] -= it->second.size; m_allocations.erase(it); } free(ptr); } // 누수 탐지: 특정 시간 이상 살아있는 할당 찾기 std::vector<AllocationRecord> FindPotentialLeaks(uint64_t ageThresholdMs) { std::vector<AllocationRecord> leaks; uint64_t now = GetCurrentTimestamp(); std::lock_guard<std::mutex> lock(m_mutex); for (const auto& [addr, record] : m_allocations) { if (now - record.timestamp > ageThresholdMs) { leaks.push_back(record); } } // 크기순 정렬 std::sort(leaks.begin(), leaks.end(), [](const auto& a, const auto& b) { return a.size > b.size; }); return leaks; } // 태그별 사용량 리포트 void PrintTaggedReport() { std::lock_guard<std::mutex> lock(m_mutex); printf("=== Memory Usage by Tag ===\n"); for (const auto& [tag, size] : m_taggedUsage) { printf(" %s: %.2f MB\n", tag, size / (1024.0 * 1024.0)); } } }; // 매크로로 편리하게 사용 #define TRACKED_ALLOC(size, tag) \ g_MemoryTracker.TrackAlloc(size, __FILE__, __LINE__, tag) #define TRACKED_FREE(ptr) \ g_MemoryTracker.TrackFree(ptr) // 사용 예 void* textureData = TRACKED_ALLOC(1024 * 1024, "Texture"); // ... 사용 후 TRACKED_FREE(textureData);

핵심 정리: 32GB RAM도 부족한 게임

  • 공간적 지역성: 동시에 모든 데이터가 필요하지 않음 - 스트리밍이 답
  • RAM vs VRAM: 분리형(PC) vs 통합형(콘솔/모바일) 아키텍처 이해 필수
  • 메모리 예산: 카테고리별 예산 배정 + 사용량 실시간 추적
  • 압박 대응: 5단계 (정상→주의→경고→위험→긴급) 단계적 대응
  • LOD/밉맵: 거리 기반 품질 조절로 메모리 90%+ 절약
  • 프로파일링: 실시간 모니터링 + 누수 탐지 + 태그별 추적
