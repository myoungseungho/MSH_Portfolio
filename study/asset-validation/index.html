에셋 검증 자동화

네이밍, 크기, 포맷 체크를 자동화하여 품질 관리

Part 1. 개념

에셋 검증 자동화가 뭐야?

에셋 검증 자동화는 프로젝트에 추가되는 모든 에셋이 정해진 규칙을 준수하는지 자동으로 검사하는 시스템입니다.

수동 검토에 의존하면 실수가 발생하기 마련입니다. 자동화된 검증 시스템은 Import 시점, 빌드 시점, CI/CD 파이프라인에서 규칙 위반을 즉시 감지하여 문제가 프로덕션까지 전파되는 것을 방지합니다.

├─────────────────────────────────────────────────────────────────────────────┤

검증 시스템의 핵심 원칙은 "Early Detection, Fast Feedback"입니다. 문제를 일찍 발견할수록 수정 비용이 낮아집니다.

Part 2. 왜 필요해?

수동 검토로는 안 돼?

소규모 프로젝트에서는 수동 검토가 가능하지만, 규모가 커지면 한계에 부딪힙니다.

┌─────────────────────────────────────────────────────────────────────────────┐
│                     수동 검토 vs 자동 검증 비교                              │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  프로젝트 규모에 따른 검증 비용:                                             │
│                                                                             │
│  비용 │                                                                    │
│   ↑   │                                            ╱ 수동 검토              │
│       │                                          ╱   (인력 비례 증가)        │
│       │                                        ╱                           │
│       │                                      ╱                             │
│       │                                    ╱                               │
│       │                                  ╱                                 │
│       │                                ╱                                   │
│       │                              ╱                                     │
│       │    ──────────────────────────────────────  자동 검증               │
│       │                              ↑             (초기 구축 후 일정)       │
│       │                              │                                     │
│       │                         손익분기점                                  │
│       │                         (약 1만 에셋)                               │
│       └──────────────────────────────────────────────────▶ 에셋 수          │
│              1천      5천      1만      5만      10만                        │
│                                                                             │
├─────────────────────────────────────────────────────────────────────────────┤
│  실제 사례:                                                                  │
│  • AAA 타이틀: 10만+ 에셋 → 수동 검토 불가능                                 │
│  • 모바일 게임: 1-5만 에셋 → 자동화 ROI 높음                                 │
│  • 인디 게임: 1천 미만 → 간단한 규칙만 자동화                                │
└─────────────────────────────────────────────────────────────────────────────┘

자동화의 진정한 가치는 일관성입니다. 사람은 피로, 실수, 주관적 판단이 개입하지만, 자동화된 시스템은 항상 동일한 기준을 적용합니다.

Part 3. Unity 구현

Unity에서 어떻게 에셋 검증을 구현해?

Unity에서는 AssetPostprocessor와 EditorWindow를 조합하여 검증 시스템을 구축합니다.

// 네이밍 규칙 검증기 using UnityEditor; using UnityEngine; using System.Text.RegularExpressions; using System.Collections.Generic; public class AssetNamingValidator : AssetPostprocessor { // 에셋 타입별 네이밍 규칙 정의 private static readonly Dictionary<string, NamingRule> _namingRules = new Dictionary<string, NamingRule> { // 텍스처: T_[Category]_[Name]_[Suffix] // Suffix: D(Diffuse), N(Normal), M(Metallic), R(Roughness), E(Emissive), O(AO) { ".png", new NamingRule(@"^T_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+_[DNMREO]$", "Texture", "T_Character_Base_D") }, { ".tga", new NamingRule(@"^T_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+_[DNMREO]$", "Texture", "T_Environment_Rock_N") }, // 메시: SM_[Category]_[Name] (Static), SK_[Category]_[Name] (Skeletal) { ".fbx", new NamingRule(@"^(SM|SK)_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+$", "Mesh", "SM_Props_Chair or SK_Character_Hero") }, // 머티리얼: M_[Category]_[Name] { ".mat", new NamingRule(@"^M_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+$", "Material", "M_Character_Skin") }, // 프리팹: P_[Category]_[Name] { ".prefab", new NamingRule(@"^P_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+$", "Prefab", "P_UI_HealthBar") }, // 애니메이션: A_[Character]_[Action] { ".anim", new NamingRule(@"^A_[A-Z][a-zA-Z0-9]+_[A-Z][a-zA-Z0-9]+$", "Animation", "A_Hero_Run") }, // 오디오: SFX_[Category]_[Name] 또는 BGM_[Name] { ".wav", new NamingRule(@"^(SFX|BGM)_[A-Z][a-zA-Z0-9_]+$", "Audio", "SFX_UI_Click or BGM_MainMenu") }, }; void OnPreprocessAsset() { string ext = System.IO.Path.GetExtension(assetPath).ToLower(); string fileName = System.IO.Path.GetFileNameWithoutExtension(assetPath); if (_namingRules.TryGetValue(ext, out NamingRule rule)) { if (!Regex.IsMatch(fileName, rule.Pattern)) { ValidationResult result = new ValidationResult { AssetPath = assetPath, Rule = "NamingConvention", Severity = ValidationSeverity.Warning, Message = $"Invalid {rule.AssetType} naming. " + $"Expected pattern: {rule.Example}" }; ValidationReporter.Report(result); } } } } public class NamingRule { public string Pattern { get; } public string AssetType { get; } public string Example { get; } public NamingRule(string pattern, string assetType, string example) { Pattern = pattern; AssetType = assetType; Example = example; } }
텍스처 크기와 포맷 검증은?

텍스처는 메모리 사용량의 주요 원인이므로 크기, 해상도, 압축 포맷을 엄격하게 검증해야 합니다.

public class TextureValidator : AssetPostprocessor { // 플랫폼별 최대 텍스처 크기 private static readonly Dictionary<BuildTarget, int> _maxTextureSize = new Dictionary<BuildTarget, int> { { BuildTarget.iOS, 2048 }, { BuildTarget.Android, 2048 }, { BuildTarget.StandaloneWindows64, 4096 }, { BuildTarget.PS5, 4096 }, }; // 경로별 최대 크기 (UI는 작게, 환경은 크게) private static readonly Dictionary<string, int> _pathSizeLimits = new Dictionary<string, int> { { "Assets/Art/UI/", 512 }, { "Assets/Art/Icons/", 256 }, { "Assets/Art/Characters/", 2048 }, { "Assets/Art/Environment/", 4096 }, }; void OnPreprocessTexture() { TextureImporter importer = (TextureImporter)assetImporter; List<ValidationResult> results = new List<ValidationResult>(); // 1. POT (Power of Two) 검증 ValidatePowerOfTwo(importer, results); // 2. 최대 크기 검증 ValidateMaxSize(importer, results); // 3. 압축 포맷 검증 ValidateCompression(importer, results); // 4. 밉맵 설정 검증 ValidateMipMaps(importer, results); // 5. Read/Write 설정 검증 ValidateReadWrite(importer, results); foreach (var result in results) { ValidationReporter.Report(result); } } private void ValidatePowerOfTwo(TextureImporter importer, List<ValidationResult> results) { // UI 스프라이트는 POT 불필요 if (importer.textureType == TextureImporterType.Sprite) return; // 원본 텍스처 크기 확인 (에디터 API 사용) int width, height; if (!GetOriginalTextureSize(assetPath, out width, out height)) return; if (!IsPowerOfTwo(width) || !IsPowerOfTwo(height)) { results.Add(new ValidationResult { AssetPath = assetPath, Rule = "PowerOfTwo", Severity = ValidationSeverity.Warning, Message = $"Texture size ({width}x{height}) is not power of two. " + "This may cause compression issues on mobile." }); } } private void ValidateCompression(TextureImporter importer, List<ValidationResult> results) { // Android 설정 확인 TextureImporterPlatformSettings androidSettings = importer.GetPlatformTextureSettings("Android"); if (androidSettings.overridden) { // ASTC 권장 (최신 기기), ETC2 허용 (호환성) var format = androidSettings.format; bool isAcceptable = format == TextureImporterFormat.ASTC_6x6 || format == TextureImporterFormat.ASTC_8x8 || format == TextureImporterFormat.ETC2_RGBA8; if (!isAcceptable) { results.Add(new ValidationResult { AssetPath = assetPath, Rule = "AndroidCompression", Severity = ValidationSeverity.Warning, Message = $"Android texture format {format} is not recommended. " + "Use ASTC_6x6/8x8 or ETC2_RGBA8." }); } } else { results.Add(new ValidationResult { AssetPath = assetPath, Rule = "PlatformOverride", Severity = ValidationSeverity.Info, Message = "No Android-specific texture settings. Consider adding platform override." }); } } private void ValidateReadWrite(TextureImporter importer, List<ValidationResult> results) { // Read/Write 활성화 = 메모리 2배 사용 if (importer.isReadable) { results.Add(new ValidationResult { AssetPath = assetPath, Rule = "ReadWriteEnabled", Severity = ValidationSeverity.Warning, Message = "Read/Write enabled doubles memory usage. " + "Disable unless runtime pixel access is needed." }); } } private bool IsPowerOfTwo(int value) { return value > 0 && (value & (value - 1)) == 0; } private bool GetOriginalTextureSize(string path, out int width, out int height) { // 실제 이미지 파일 크기 읽기 try { byte[] data = System.IO.File.ReadAllBytes(path); Texture2D temp = new Texture2D(1, 1); temp.LoadImage(data); width = temp.width; height = temp.height; Object.DestroyImmediate(temp); return true; } catch { width = height = 0; return false; } } }
메시 폴리곤 수 검증은 어떻게 해?

메시 폴리곤 수는 렌더링 성능에 직접 영향을 미칩니다. LOD 레벨별, 용도별로 다른 기준을 적용합니다.

public class MeshValidator : AssetPostprocessor { // 메시 카테고리별 폴리곤 제한 private static readonly Dictionary<string, MeshLimits> _meshLimits = new Dictionary<string, MeshLimits> { // 메인 캐릭터 (높은 품질) { "Characters/Hero", new MeshLimits(30000, 15000, 5000, 1000) }, // NPC/몬스터 (중간 품질) { "Characters/NPC", new MeshLimits(15000, 7500, 3000, 500) }, // 환경 오브젝트 (다양) { "Environment/Props", new MeshLimits(5000, 2500, 1000, 200) }, // 배경 건물 { "Environment/Buildings", new MeshLimits(50000, 20000, 5000, 1000) }, // 무기/장비 { "Items/Weapons", new MeshLimits(8000, 4000, 1500, 300) }, }; void OnPostprocessModel(GameObject model) { MeshFilter[] meshFilters = model.GetComponentsInChildren<MeshFilter>(); SkinnedMeshRenderer[] skinnedMeshes = model.GetComponentsInChildren<SkinnedMeshRenderer>(); int totalVertices = 0; int totalTriangles = 0; // 정적 메시 합산 foreach (var mf in meshFilters) { if (mf.sharedMesh != null) { totalVertices += mf.sharedMesh.vertexCount; totalTriangles += mf.sharedMesh.triangles.Length / 3; } } // 스킨드 메시 합산 foreach (var smr in skinnedMeshes) { if (smr.sharedMesh != null) { totalVertices += smr.sharedMesh.vertexCount; totalTriangles += smr.sharedMesh.triangles.Length / 3; } } // 카테고리 판별 MeshLimits limits = GetLimitsForPath(assetPath); if (totalTriangles > limits.LOD0Max) { ValidationReporter.Report(new ValidationResult { AssetPath = assetPath, Rule = "MeshPolygonCount", Severity = ValidationSeverity.Error, Message = $"Mesh has {totalTriangles:N0} triangles. " + $"Maximum allowed for this category: {limits.LOD0Max:N0}" }); } // LOD 존재 여부 확인 LODGroup lodGroup = model.GetComponent<LODGroup>(); if (lodGroup == null && totalTriangles > 3000) { ValidationReporter.Report(new ValidationResult { AssetPath = assetPath, Rule = "MissingLOD", Severity = ValidationSeverity.Warning, Message = $"High-poly mesh ({totalTriangles:N0} tris) without LOD. " + "Consider adding LOD levels for performance." }); } // 본 개수 검증 (모바일 제한) if (skinnedMeshes.Length > 0) { int maxBones = skinnedMeshes.Max(smr => smr.sharedMesh?.bindposes?.Length ?? 0); if (maxBones > 75) // 모바일 권장 최대 { ValidationReporter.Report(new ValidationResult { AssetPath = assetPath, Rule = "BoneCount", Severity = ValidationSeverity.Warning, Message = $"Skinned mesh has {maxBones} bones. " + "Mobile recommends ≤75 bones per mesh." }); } } } private MeshLimits GetLimitsForPath(string path) { foreach (var kvp in _meshLimits) { if (path.Contains(kvp.Key)) return kvp.Value; } // 기본 제한 return new MeshLimits(10000, 5000, 2000, 500); } } public struct MeshLimits { public int LOD0Max; // 가장 높은 품질 public int LOD1Max; // 중간 거리 public int LOD2Max; // 먼 거리 public int LOD3Max; // 가장 먼 거리 public MeshLimits(int lod0, int lod1, int lod2, int lod3) { LOD0Max = lod0; LOD1Max = lod1; LOD2Max = lod2; LOD3Max = lod3; } }

Part 4. 통합 검증 시스템

모든 검증을 통합 관리하는 시스템은?

개별 검증기를 중앙 집중식 리포터로 통합하면 일관된 결과 수집과 리포팅이 가능합니다.

using System; using System.Collections.Generic; using System.IO; using UnityEditor; using UnityEngine; public enum ValidationSeverity { Info, // 정보성 - 로그만 Warning, // 경고 - 리포트 생성 Error // 에러 - Import 차단 가능 } public class ValidationResult { public string AssetPath { get; set; } public string Rule { get; set; } public ValidationSeverity Severity { get; set; } public string Message { get; set; } public DateTime Timestamp { get; set; } = DateTime.Now; public string ToLogString() { return $"[{Severity}] {Rule}: {AssetPath}\n {Message}"; } public string ToCsvLine() { return $"\"{Timestamp:yyyy-MM-dd HH:mm:ss}\",\"{Severity}\",\"{Rule}\",\"{AssetPath}\",\"{Message}\""; } } public static class ValidationReporter { private static List<ValidationResult> _sessionResults = new List<ValidationResult>(); private static readonly string ReportPath = "Assets/Editor/ValidationReports/"; // 검증 설정 public static bool BlockOnError { get; set; } = false; public static bool LogToConsole { get; set; } = true; public static bool AutoExportReport { get; set; } = true; public static void Report(ValidationResult result) { _sessionResults.Add(result); if (LogToConsole) { switch (result.Severity) { case ValidationSeverity.Info: Debug.Log(result.ToLogString()); break; case ValidationSeverity.Warning: Debug.LogWarning(result.ToLogString()); break; case ValidationSeverity.Error: Debug.LogError(result.ToLogString()); break; } } // Error 레벨에서 Import 차단 if (BlockOnError && result.Severity == ValidationSeverity.Error) { throw new Exception( $"Asset validation failed: {result.Message}"); // CI용 배치 모드 검증 러너 using UnityEditor; using UnityEngine; using System; using System.IO; using System.Collections.Generic; public static class CIValidationRunner { private const string OutputPath = "./validation_report.json"; // CI에서 호출: Unity -batchmode -executeMethod CIValidationRunner.Run public static void Run() { Debug.Log("=== Starting CI Asset Validation ==="); List<ValidationResult> allResults = new List<ValidationResult>(); bool hasErrors = false; try { // 1. 모든 에셋 수집 string[] assetPaths = AssetDatabase.GetAllAssetPaths(); int totalAssets = assetPaths.Length; int processed = 0; foreach (string path in assetPaths) { if (!path.StartsWith("Assets/")) continue; // 각 에셋 검증 var results = ValidateAsset(path); allResults.AddRange(results); if (results.Any(r => r.Severity == ValidationSeverity.Error)) hasErrors = true; processed++; if (processed % 500 == 0) Debug.Log($"Progress: {processed}/{totalAssets}"); } // 2. 리포트 생성 GenerateReport(allResults); // 3. 요약 출력 int errors = allResults.Count(r => r.Severity == ValidationSeverity.Error); int warnings = allResults.Count(r => r.Severity == ValidationSeverity.Warning); Debug.Log($"\n=== Validation Complete ==="); Debug.Log($"Total Assets: {totalAssets}"); Debug.Log($"Errors: {errors}"); Debug.Log($"Warnings: {warnings}"); Debug.Log($"Report: {OutputPath}"); // 4. 에러가 있으면 실패 코드로 종료 if (hasErrors) { Debug.LogError("Validation FAILED - Errors found"); EditorApplication.Exit(1); } else { Debug.Log("Validation PASSED"); EditorApplication.Exit(0); } } catch (Exception ex) { Debug.LogError($"Validation crashed: {ex}"); EditorApplication.Exit(2); } } private static List<ValidationResult> ValidateAsset(string path) { var results = new List<ValidationResult>(); // 확장자별 검증 로직 분기 string ext = Path.GetExtension(path).ToLower(); switch (ext) { case ".png": case ".tga": case ".jpg": results.AddRange(TextureValidationRules.Validate(path)); break; case ".fbx": case ".obj": results.AddRange(MeshValidationRules.Validate(path)); break; case ".prefab": results.AddRange(PrefabValidationRules.Validate(path)); break; } // 공통 규칙 (네이밍) results.AddRange(NamingValidationRules.Validate(path)); return results; } private static void GenerateReport(List<ValidationResult> results) { var report = new { timestamp = DateTime.UtcNow.ToString("o"), summary = new { total = results.Count, errors = results.Count(r => r.Severity == ValidationSeverity.Error), warnings = results.Count(r => r.Severity == ValidationSeverity.Warning), info = results.Count(r => r.Severity == ValidationSeverity.Info) }, issues = results.Select(r => new { severity = r.Severity.ToString(), rule = r.Rule, path = r.AssetPath, message = r.Message }) }; string json = JsonUtility.ToJson(report, true); File.WriteAllText(OutputPath, json); } }
