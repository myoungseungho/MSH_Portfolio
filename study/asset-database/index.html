Part 1. 개념 - 에셋 데이터베이스란?

에셋 데이터베이스가 정확히 뭐야?

에셋 데이터베이스(Asset Database)는 게임 엔진이 프로젝트의 모든 에셋을 추적, 인덱싱, 관리하는 내부 시스템이야.

핵심 역할:

  • 에셋 인덱싱: 모든 에셋의 위치, 타입, 메타데이터 기록
  • 의존성 추적: 에셋 간 참조 관계 관리
  • 임포트 캐시: 변환된 에셋 캐시로 재임포트 방지
  • 고유 식별자: GUID로 에셋 이동/이름변경에도 참조 유지

Library 폴더 주의사항

절대로 버전 관리에 포함하지 말 것!

  • 용량이 수십 GB에 달할 수 있음

Part 2. GUID 시스템

GUID가 왜 중요해?

GUID (Globally Unique Identifier)는 에셋 참조의 핵심이야. 파일 경로 대신 고유 ID로 에셋을 식별해:

// Unity .meta 파일 예시 (hero.png.meta) fileFormatVersion: 2 guid: abc123def456789012345678 ← 고유 식별자 TextureImporter: internalIDToNameTable: [] externalObjects: {} serializedVersion: 12 mipmaps: mipMapMode: 0 enableMipMap: 1 textureSettings: filterMode: 1 aniso: 1 wrapU: 0 wrapV: 0 maxTextureSize: 2048 textureFormat: -1 ← 자동 포맷 compressionQuality: 50

GUID의 장점:

  • 파일 이름 변경해도 참조 유지
  • 폴더 이동해도 참조 유지
  • 프로젝트 전체에서 유일성 보장

GUID가 깨지는 경우

1. .meta 파일 삭제

  • 새 GUID 생성 → 모든 참조 끊김
  • .meta 파일은 반드시 소스 파일과 함께 이동/커밋

2. 외부 도구로 파일 이동

  • Windows 탐색기로 이동 시 .meta 누락 가능
  • 반드시 Unity Editor 내에서 이동

3. Git 머지 충돌

  • 같은 GUID로 다른 에셋이 생성된 경우
  • GUID 충돌 해결 도구 필요
GUID 충돌은 어떻게 감지하고 해결해?

GUID 충돌은 심각한 문제야. 두 에셋이 같은 ID를 가지면 하나만 인식돼:

// Unity GUID 충돌 감지 스크립트 using UnityEditor; using UnityEngine; using System.Collections.Generic; using System.IO; public static class GUIDConflictDetector { [MenuItem("Tools/Detect GUID Conflicts")] public static void DetectConflicts() { var guidToPath = new Dictionary<string, List<string>>(); var metaFiles = Directory.GetFiles( "Assets", "*.meta", SearchOption.AllDirectories); foreach (var metaPath in metaFiles) { var content = File.ReadAllText(metaPath); var guidMatch = System.Text.RegularExpressions.Regex.Match( content, @"guid:\s*([a-f0-9]+)"); if (guidMatch.Success) { var guid = guidMatch.Groups[1].Value; if (!guidToPath.ContainsKey(guid)) guidToPath[guid] = new List<string>(); guidToPath[guid].Add(metaPath); } } // 충돌 보고 var conflicts = 0; foreach (var kvp in guidToPath) { if (kvp.Value.Count > 1) { conflicts++; Debug.LogError($"GUID 충돌 발견: {kvp.Key}"); foreach (var path in kvp.Value) Debug.LogError($" - {path}"); } } if (conflicts == 0) Debug.Log("GUID 충돌 없음!"); else Debug.LogError($"총 {conflicts}개의 GUID 충돌 발견!"); } [MenuItem("Tools/Regenerate GUID for Selected")] public static void RegenerateGUID() { var selected = Selection.assetGUIDs; foreach (var guid in selected) { var path = AssetDatabase.GUIDToAssetPath(guid); var metaPath = path + ".meta"; if (!File.Exists(metaPath)) continue; var content = File.ReadAllText(metaPath); var newGuid = GUID.Generate().ToString().Replace("-", ""); content = System.Text.RegularExpressions.Regex.Replace( content, @"guid:\s*[a-f0-9]+", $"guid: {newGuid}"); File.WriteAllText(metaPath, content); Debug.Log($"GUID 재생성: {path} → {newGuid}"); } AssetDatabase.Refresh(); } }
Unreal의 에셋 식별 방식은 달라?

Unreal은 GUID 대신 Object Path와 Package GUID를 조합해서 사용해:

// Unreal 에셋 경로 시스템 // 에셋 참조 방식 FSoftObjectPath AssetPath = "/Game/Characters/Hero/SK_Hero.SK_Hero"; // ├─── 패키지 경로 ───┤├─ 오브젝트명 ─┤ // 구성요소 "/Game" ← Content 폴더 루트 "/Characters/Hero" ← 폴더 경로 "/SK_Hero" ← 패키지 (uasset 파일) ".SK_Hero" ← 오브젝트 (패키지 내 에셋) // 참조 타입 TSoftObjectPtr<UTexture2D> ← 소프트 참조 (지연 로드) TObjectPtr<UTexture2D> ← 하드 참조 (즉시 로드) // 코드에서 참조 UTexture2D* Texture = LoadObject<UTexture2D>( nullptr, TEXT("/Game/Textures/T_Hero_Diffuse.T_Hero_Diffuse") ); // 블루프린트에서 참조 FSoftObjectPath Path; Path.SetPath("/Game/Characters/Hero/BP_Hero.BP_Hero_C");

Unreal 에셋 이동 주의

Unreal은 경로 기반이므로 이동 시 Redirector가 생성됨

  • 이동한 에셋의 원래 위치에 Redirector 생성
  • 다른 에셋들이 Redirector를 통해 새 위치 찾음
  • Fix Up Redirectors로 정리 필요

AssetPostprocessor로 임포트를 커스터마이징하는 방법은?

Unity의 AssetPostprocessor를 사용하면 임포트 동작을 세밀하게 제어할 수 있어:

// 커스텀 에셋 임포트 프로세서 using UnityEditor; using UnityEngine; public class CustomAssetProcessor : AssetPostprocessor { // ═══ 텍스처 임포트 ═══ void OnPreprocessTexture() { var importer = (TextureImporter)assetImporter; // 폴더 기반 자동 설정 if (assetPath.Contains("/UI/")) { importer.textureType = TextureImporterType.Sprite; importer.mipmapEnabled = false; importer.filterMode = FilterMode.Point; } else if (assetPath.Contains("/NormalMaps/")) { importer.textureType = TextureImporterType.NormalMap; importer.mipmapEnabled = true; } else if (assetPath.Contains("/Lightmaps/")) { importer.textureType = TextureImporterType.Lightmap; importer.mipmapEnabled = false; } // 네이밍 컨벤션 기반 설정 if (assetPath.Contains("_Normal")) importer.textureType = TextureImporterType.NormalMap; else if (assetPath.Contains("_Mask")) importer.sRGBTexture = false; } void OnPostprocessTexture(Texture2D texture) { // 임포트 후 처리 Debug.Log($"텍스처 임포트 완료: {assetPath}"); Debug.Log($" 크기: {texture.width}x{texture.height}"); Debug.Log($" 포맷: {texture.format}"); } // ═══ 모델 임포트 ═══ void OnPreprocessModel() { var importer = (ModelImporter)assetImporter; // 공통 설정 importer.importNormals = ModelImporterNormals.Import; importer.importTangents = ModelImporterTangents.CalculateMikk; // 애니메이션 있는 모델 if (assetPath.Contains("/Characters/")) { importer.animationType = ModelImporterAnimationType.Human; importer.avatarSetup = ModelImporterAvatarSetup.CreateFromThisModel; } // 정적 모델 else if (assetPath.Contains("/Props/")) { importer.animationType = ModelImporterAnimationType.None; importer.isReadable = false; ← 메모리 최적화 } } void OnPostprocessModel(GameObject model) { // LOD 그룹 자동 설정 if (assetPath.Contains("_LOD")) { SetupLODGroup(model); } } // ═══ 오디오 임포트 ═══ void OnPreprocessAudio() { var importer = (AudioImporter)assetImporter; if (assetPath.Contains("/Music/")) { // 음악: 스트리밍 importer.loadInBackground = true; var settings = importer.defaultSampleSettings; settings.loadType = AudioClipLoadType.Streaming; importer.defaultSampleSettings = settings; } else if (assetPath.Contains("/SFX/")) { // 효과음: 압축 메모리 var settings = importer.defaultSampleSettings; settings.loadType = AudioClipLoadType.CompressedInMemory; importer.defaultSampleSettings = settings; } } // ═══ 모든 에셋 공통 ═══ void OnPostprocessAllAssets( string[] importedAssets, string[] deletedAssets, string[] movedAssets, string[] movedFromPaths) { // 에셋 변경 로깅 if (importedAssets.Length > 0) Debug.Log($"임포트: {importedAssets.Length}개"); if (deletedAssets.Length > 0) Debug.Log($"삭제: {deletedAssets.Length}개"); if (movedAssets.Length > 0) Debug.Log($"이동: {movedAssets.Length}개"); } }

// Library 폴더 복구 스크립트 using UnityEditor; using System.IO; public static class LibraryRecovery { [MenuItem("Tools/Recovery/Rebuild Asset Database")] public static void RebuildAssetDatabase() { // 전체 재임포트 AssetDatabase.Refresh(ImportAssetOptions.ForceUpdate); Debug.Log("에셋 데이터베이스 재구축 완료"); } [MenuItem("Tools/Recovery/Clear Script Cache")] public static void ClearScriptCache() { var scriptAssembliesPath = "Library/ScriptAssemblies"; if (Directory.Exists(scriptAssembliesPath)) { Directory.Delete(scriptAssembliesPath, true); Debug.Log("스크립트 캐시 삭제 완료. 에디터 재시작 필요."); } } [MenuItem("Tools/Recovery/Reimport All Textures")] public static void ReimportAllTextures() { var textures = AssetDatabase.FindAssets("t:Texture"); Debug.Log($"텍스처 {textures.Length}개 재임포트 시작..."); AssetDatabase.StartAssetEditing(); try { foreach (var guid in textures) { var path = AssetDatabase.GUIDToAssetPath(guid); AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceUpdate); } } finally { AssetDatabase.StopAssetEditing(); } Debug.Log("텍스처 재임포트 완료"); } }

// Addressables와 AssetDatabase 연동 using UnityEditor; using UnityEditor.AddressableAssets; using UnityEditor.AddressableAssets.Settings; public static class AddressablesIntegration { // 에셋을 Addressables에 자동 등록 [MenuItem("Tools/Addressables/Auto Register Assets")] public static void AutoRegisterAssets() { var settings = AddressableAssetSettingsDefaultObject.Settings; var defaultGroup = settings.DefaultGroup; // Characters 폴더의 모든 프리팹 등록 var guids = AssetDatabase.FindAssets( "t:Prefab", new[] { "Assets/Characters" }); foreach (var guid in guids) { var path = AssetDatabase.GUIDToAssetPath(guid); var entry = settings.CreateOrMoveEntry(guid, defaultGroup); // 주소 설정 (폴더 경로를 주소로) entry.address = path.Replace("Assets/", "") .Replace(".prefab", ""); // 라벨 추가 entry.SetLabel("Characters", true); Debug.Log($"등록: {entry.address}"); } settings.SetDirty(AddressableAssetSettings.ModificationEvent.EntryMoved, null, true); } // 에셋 변경 시 자동 Addressables 업데이트 public class AddressablesAutoUpdater : AssetPostprocessor { static void OnPostprocessAllAssets( string[] importedAssets, string[] deletedAssets, string[] movedAssets, string[] movedFromAssetPaths) { foreach (var path in importedAssets) { if (ShouldAutoRegister(path)) { RegisterToAddressables(path); } } } static bool ShouldAutoRegister(string path) { // Addressables 자동 등록 조건 return path.StartsWith("Assets/Bundles/") || path.StartsWith("Assets/DLC/"); } static void RegisterToAddressables(string path) { var guid = AssetDatabase.AssetPathToGUID(path); var settings = AddressableAssetSettingsDefaultObject.Settings; if (settings.FindAssetEntry(guid) == null) { var group = settings.DefaultGroup; settings.CreateOrMoveEntry(guid, group); Debug.Log($"[Addressables] 자동 등록: {path}"); } } } }
