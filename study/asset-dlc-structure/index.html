DLC 구조 설계

기본 게임과 추가 컨텐츠의 분리, 버전 호환성, 플랫폼별 배포

Part 1. 개념: DLC란 무엇인가?

DLC(Downloadable Content)가 뭐고 왜 필요해?

DLC는 기본 게임 출시 후 추가로 제공되는 컨텐츠입니다. 새로운 캐릭터, 스토리, 맵, 아이템 등을 별도로 판매하거나 무료로 배포합니다.

DLC의 비즈니스 가치:

// Addressables DLC 그룹 설정 public class DLCBundleSetup { [MenuItem("DLC/Setup Addressable Groups")] public static void SetupDLCGroups() { var settings = AddressableAssetSettingsDefaultObject.Settings; // 1. 공유 에셋 그룹 (Base + DLC 공용) var sharedGroup = CreateGroup(settings, "Shared_Common", new BundledAssetGroupSchema { BundleMode = BundledAssetGroupSchema.BundlePackingMode.PackTogether, BuildPath = "ServerData/[BuildTarget]/shared", LoadPath = "{DLC.SharedPath}/[BuildTarget]/shared" }); // 공유 에셋 등록 var sharedAssets = AssetDatabase.FindAssets("t:Material", new[] { "Assets/_Base/Common" }); foreach (var guid in sharedAssets) { var path = AssetDatabase.GUIDToAssetPath(guid); var entry = settings.CreateOrMoveEntry(guid, sharedGroup); entry.address = "shared/" + Path.GetFileNameWithoutExtension(path); } // 2. DLC별 그룹 생성 var dlcFolders = Directory.GetDirectories("Assets/_DLC"); foreach (var dlcFolder in dlcFolders) { var dlcName = Path.GetFileName(dlcFolder); var dlcGroup = CreateGroup(settings, dlcName, new BundledAssetGroupSchema { BundleMode = BundledAssetGroupSchema.BundlePackingMode.PackSeparately, BuildPath = $"ServerData/[BuildTarget]/{dlcName.ToLower()}", LoadPath = $"{{DLC.{dlcName}Path}}/[BuildTarget]" }); // DLC 에셋 등록 (공유 에셋 제외) var dlcAssets = AssetDatabase.FindAssets("", new[] { dlcFolder }); foreach (var guid in dlcAssets) { var path = AssetDatabase.GUIDToAssetPath(guid); // 이미 공유 그룹에 있으면 스킵 if (IsInSharedGroup(settings, guid)) continue; var entry = settings.CreateOrMoveEntry(guid, dlcGroup); entry.address = $"{dlcName.ToLower()}/{Path.GetFileNameWithoutExtension(path)}"; } } } }

// 플랫폼별 소유권 확인 시스템 public class DLCEntitlementManager { private IDLCPlatformProvider platformProvider; private Dictionary<string, DLCEntitlement> entitlements = new(); private const string CACHE_KEY = "dlc_entitlements"; public async Task InitializeAsync() { // 플랫폼별 프로바이더 선택 #if UNITY_STANDALONE_WIN platformProvider = new SteamDLCProvider(); #elif UNITY_PS5 platformProvider = new PlayStationDLCProvider(); #elif UNITY_GAMECORE platformProvider = new XboxDLCProvider(); #elif UNITY_IOS || UNITY_ANDROID platformProvider = new UnityIAPDLCProvider(); #endif // 1. 로컬 캐시 로드 (오프라인 대비) LoadCachedEntitlements(); // 2. 온라인이면 서버에서 갱신 if (Application.internetReachability != NetworkReachability.NotReachable) { await RefreshEntitlementsAsync(); } } public async Task RefreshEntitlementsAsync() { try { var ownedDLCs = await platformProvider.GetOwnedDLCsAsync(); foreach (var dlc in ownedDLCs) { entitlements[dlc.dlcId] = new DLCEntitlement { dlcId = dlc.dlcId, owned = true, purchaseDate = dlc.purchaseDate, lastVerified = DateTime.UtcNow }; } SaveCachedEntitlements(); } catch (Exception e) { Debug.LogWarning($"Failed to refresh entitlements: {e.Message}"); // 캐시 데이터 사용 } } public bool IsDLCOwned(string dlcId) { return entitlements.TryGetValue(dlcId, out var ent) && ent.owned; } public IEnumerable<string> GetOwnedDLCIds() { return entitlements.Where(e => e.Value.owned).Select(e => e.Key); } void LoadCachedEntitlements() { var json = PlayerPrefs.GetString(CACHE_KEY, ""); if (!string.IsNullOrEmpty(json)) { entitlements = JsonUtility.FromJson<Dictionary<string, DLCEntitlement>>(json); } } void SaveCachedEntitlements() { var json = JsonUtility.ToJson(entitlements); PlayerPrefs.SetString(CACHE_KEY, json); PlayerPrefs.Save(); } } // Steam 구현 예시 public class SteamDLCProvider : IDLCPlatformProvider { public async Task<List<OwnedDLCInfo>> GetOwnedDLCsAsync() { var result = new List<OwnedDLCInfo>(); var dlcCount = SteamApps.GetDLCCount(); for (int i = 0; i < dlcCount; i++) { if (SteamApps.BGetDLCDataByIndex(i, out var appId, out var available, out var name)) { if (SteamApps.BIsDlcInstalled(appId)) { result.Add(new OwnedDLCInfo { dlcId = AppIdToDlcId(appId), purchaseDate = DateTime.UtcNow // Steam doesn't provide this }); } } } return result; } }
구매하지 않은 DLC 컨텐츠는 어떻게 처리해?

미구매 DLC는 잠금 상태로 표시하고 구매 유도합니다:

미구매 DLC 처리 전략:

1. UI에서 잠금 표시
   ┌─────────────────────────────────────────────────────────────┐
   │  캐릭터 선택                                                 │
   │  ┌─────┐  ┌─────┐  ┌─────────────────────┐                 │
   │  │     │  │     │  │ 🔒                  │                 │
   │  │ 🧙  │  │ 🗡️  │  │    닌자             │                 │
   │  │     │  │     │  │                     │                 │
   │  │Hero │  │Warrior│ │ "시즌 1 DLC 필요"   │                 │
   │  └─────┘  └─────┘  │                     │                 │
   │  [선택]   [선택]    │    [구매하기]        │                 │
   │                    └─────────────────────┘                 │
   └─────────────────────────────────────────────────────────────┘

2. 인게임 프리뷰 제공
   ┌─────────────────────────────────────────────────────────────┐
   │  - 미구매 캐릭터의 스킬 영상 미리보기                         │
   │  - 미구매 맵의 스크린샷/트레일러                              │
   │  - 체험판 스테이지 제공 (첫 번째 미션만)                      │
   │  → 구매 전환율 증가                                         │
   └─────────────────────────────────────────────────────────────┘

3. Fallback 처리 (코드 레벨)
   ┌─────────────────────────────────────────────────────────────┐
   │  DLC 캐릭터 로드 시도                                        │
   │      │                                                      │
   │      ▼                                                      │
   │  소유권 확인 ────────→ NO ────→ 기본 캐릭터로 대체            │
   │      │                         + 구매 팝업 표시              │

// DLC 버전 호환성 검사기 public class DLCCompatibilityChecker { private readonly string gameVersion; public DLCCompatibilityChecker(string currentGameVersion) { gameVersion = currentGameVersion; } public CompatibilityResult CheckDLC(DLCManifest manifest) { var current = new Version(gameVersion); var min = new Version(manifest.minGameVersion); var max = new Version(manifest.maxGameVersion); if (current < min) { return new CompatibilityResult { isCompatible = false, reason = CompatibilityReason.GameTooOld, message = $"게임 버전이 너무 낮습니다. {min} 이상 필요.", requiredAction = "게임을 업데이트해주세요." }; } if (current >= max) { return new CompatibilityResult { isCompatible = false, reason = CompatibilityReason.DLCTooOld, message = $"DLC 버전이 현재 게임과 호환되지 않습니다.", requiredAction = "DLC 업데이트를 확인해주세요." }; } return new CompatibilityResult { isCompatible = true }; } public List<DLCManifest> FilterCompatibleDLCs( IEnumerable<DLCManifest> manifests) { var compatible = new List<DLCManifest>(); var incompatible = new List<(DLCManifest, CompatibilityResult)>(); foreach (var manifest in manifests) { var result = CheckDLC(manifest); if (result.isCompatible) { compatible.Add(manifest); } else { incompatible.Add((manifest, result)); Debug.LogWarning( $"DLC '{manifest.dlcId}' 비호환: {result.message}"); } } // 비호환 DLC 알림 if (incompatible.Count > 0) { OnIncompatibleDLCsFound?.Invoke(incompatible); } return compatible; } public event Action<List<(DLCManifest, CompatibilityResult)>> OnIncompatibleDLCsFound; } public enum CompatibilityReason { Compatible, GameTooOld, // 게임 업데이트 필요 DLCTooOld, // DLC 업데이트 필요 MissingDep, // 필수 DLC 누락 Incompatible // 상호 배타적 }

Manifest 로드 var manifest = await LoadManifestAsync(dlcId); // 2. 이미 완료된 경우 스킵 if (GetStatus(dlcId) == DownloadStatus.Complete) { Debug.Log($"DLC '{dlcId}' already downloaded."); return; } downloadStatus[dlcId] = DownloadStatus.Downloading; try { long totalSize = manifest.totalSize; long downloadedSize = 0; foreach (var bundle in manifest.bundles) { ct.ThrowIfCancellationRequested(); var localPath = Path.Combine(localDLCPath, dlcId, bundle.name); // 이미 다운로드된 파일 확인 (재시작 지원) if (File.Exists(localPath) && VerifyHash(localPath, bundle.hash)) { downloadedSize += bundle.size; continue; } // 다운로드 var url = $"{cdnBaseUrl}/{dlcId}/{bundle.name}"; await DownloadFileAsync(url, localPath, bundle.size, new Progress<long>(bytes => { var currentTotal = downloadedSize + bytes; progress?.Report(new DownloadProgress { dlcId = dlcId, currentBundle = bundle.name, downloadedBytes = currentTotal, totalBytes = totalSize, progress = (float)currentTotal / totalSize }); }), ct); // 해시 검증 if (!VerifyHash(localPath, bundle.hash)) { File.Delete(localPath); throw new Exception($"Hash mismatch for {bundle.name}"); } downloadedSize += bundle.size; } // 완료 처리 downloadStatus[dlcId] = DownloadStatus.Complete; SaveDownloadStatus(); // Addressables 카탈로그 업데이트 await UpdateAddressablesCatalogAsync(dlcId); OnDLCDownloadComplete?.Invoke(dlcId); } catch (OperationCanceledException) { downloadStatus[dlcId] = DownloadStatus.Paused; SaveDownloadStatus(); throw; } catch (Exception e) { downloadStatus[dlcId] = DownloadStatus.Failed; SaveDownloadStatus(); Debug.LogError($"DLC download failed: {e.Message}"); throw; } } public async Task DeleteDLCAsync(string dlcId) { var dlcPath = Path.Combine(localDLCPath, dlcId); if (Directory.Exists(dlcPath)) { Directory.Delete(dlcPath, true); } downloadStatus[dlcId] = DownloadStatus.NotStarted; SaveDownloadStatus(); // Addressables에서 제거 await Addressables.ClearDependencyCacheAsync(dlcId); OnDLCDeleted?.Invoke(dlcId); } public DownloadStatus GetStatus(string dlcId) { return downloadStatus.TryGetValue(dlcId, out var status) ? status : DownloadStatus.NotStarted; Part 7. 실전 패턴과 주의사항

DLC 설계 시 흔한 실수는?

⚠️ 피해야 할 안티패턴

  • Base → DLC 참조: 기본 게임이 DLC를 직접 참조하면 DLC 없이 실행 불가
  • DLC 간 암묵적 의존성: manifest에 명시하지 않은 DLC 간 참조
  • 공유 에셋 중복: 같은 Material이 여러 DLC에 포함되어 용량 낭비
  • 버전 범위 누락: minGameVersion 없이 배포하면 구버전에서 크래시
  • 오프라인 미고려: 항상 서버 확인하면 오프라인 플레이 불가

✅ 권장 패턴

  • Composition 패턴: DLC는 기본 게임 시스템에 "등록"만 하는 방식
  • Fallback 준비: DLC 에셋 로드 실패 시 기본 에셋으로 대체
  • 점진적 다운로드: 필수 번들 먼저, 선택 번들은 나중에
  • 백그라운드 다운로드: 플레이 중에도 DLC 다운로드 진행
  • 공간 관리 UI: 사용자가 DLC를 삭제하여 공간 확보 가능

// DLC Registry 패턴 구현 public class DLCRegistry : MonoBehaviour { public static DLCRegistry Instance { get; private set; } private Dictionary<string, IDLCContent> characters = new(); private Dictionary<string, IDLCContent> weapons = new(); private Dictionary<string, IDLCContent> maps = new(); void Awake() { Instance = this; } public void RegisterCharacter(string id, IDLCContent content) { characters[id] = content; OnContentRegistered?.Invoke(ContentType.Character, id); } public void UnregisterCharacter(string id) { characters.Remove(id); OnContentUnregistered?.Invoke(ContentType.Character, id); } public T GetCharacter<T>(string id) where T : class, IDLCContent { return characters.TryGetValue(id, out var content) ? content as T : null; } public IEnumerable<string> GetAvailableCharacterIds() { return characters.Keys; } public event Action<ContentType, string> OnContentRegistered; public event Action<ContentType, string> OnContentUnregistered; } // DLC 진입점 (각 DLC 번들에 포함) public class DLCSeason1Initializer : MonoBehaviour { [SerializeField] private CharacterData ninjaCharacter; [SerializeField] private WeaponData katanaWeapon; void OnEnable() { // DLC 로드 시 자동 등록 DLCRegistry.Instance.RegisterCharacter("ninja", ninjaCharacter); DLCRegistry.Instance.RegisterWeapon("katana", katanaWeapon); Debug.Log("DLC Season 1 activated!"); } void OnDisable() { // DLC 언로드 시 등록 해제 DLCRegistry.Instance.UnregisterCharacter("ninja"); DLCRegistry.Instance.UnregisterWeapon("katana"); } } // UI에서 사용 public class CharacterSelectUI : MonoBehaviour { void Start() { DLCRegistry.Instance.OnContentRegistered += OnContentAdded; RefreshCharacterList(); } void RefreshCharacterList() { // Base 캐릭터 + 등록된 DLC 캐릭터 모두 표시 var allCharacters = baseCharacters .Concat(DLCRegistry.Instance.GetAvailableCharacterIds()); foreach (var id in allCharacters) { CreateCharacterButton(id); } } void OnContentAdded(ContentType type, string id) { if (type == ContentType.Character) { CreateCharacterButton(id); ShowNewContentNotification(id); } } }

핵심 정리

  • 단방향 의존성 - DLC → Base 참조만 허용, Base → DLC 금지
  • 공유 에셋 분리 - 중복 방지를 위해 shared 번들 생성
  • Manifest 관리 - 버전 범위, 의존성, 번들 목록 명시
  • 소유권 확인 - 플랫폼 API + 로컬 캐시 조합
  • 버전 호환성 - minGameVersion/maxGameVersion 필수
  • 플랫폼 추상화 - IDLCPlatformProvider 인터페이스
  • 점진적 다운로드 - 재시작 지원, 해시 검증
  • Registry 패턴 - DLC가 스스로 등록하는 느슨한 결합
