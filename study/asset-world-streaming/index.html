월드 스트리밍 - 오픈월드의 비밀

100GB 월드를 16GB RAM에서 돌리는 마법

Part 1. 오픈월드의 기술적 도전

GTA V 맵이 80GB인데 8GB RAM PC에서 어떻게 돌아가?

핵심 원리: 전체를 로딩하지 않는다

플레이어가 볼 수 있고, 갈 수 있는 영역만 메모리에 올립니다.

나머지는 디스크에 두고, 필요할 때 스트리밍합니다.

월드 스트리밍과 일반 레벨 로딩의 차이는?

전통적 레벨 로딩:

• 레벨 A → 로딩 화면 → 레벨 B (전체 교체)

• 경계가 명확함 (문, 엘리베이터 등)

월드 스트리밍:

• 플레이어 이동에 따라 점진적으로 로드/언로드

• 로딩 화면 없이 심리스(Seamless) 전환

• 항상 일부만 메모리에, 나머지는 대기 상태

Part 2. 청크(Chunk) 기반 월드 분할

월드를 어떻게 나눠?

월드를 청크(Chunk) 또는 셀(Cell) 단위로 분할합니다.

각 청크는 독립적으로 로드/언로드 가능한 단위입니다.

Part 3. 스트리밍 반경과 히스테리시스

어느 거리까지 로딩해야 해?

스트리밍 반경은 게임마다, 상황마다 다릅니다:

• 시야 거리: 플레이어가 볼 수 있는 최대 거리

• 이동 속도: 빠르면 더 멀리 로딩 필요

• 메모리 여유: 남는 메모리만큼 더 로딩 가능

// 스트리밍 반경 설정 struct StreamingConfig { // 기본 반경 (미터) float loadRadius = 100.0f; // 이 안에 들어오면 로드 시작 float unloadRadius = 200.0f; // 이 밖으로 나가면 언로드 float highDetailRadius = 50.0f; // 고품질 에셋 범위 // LOD 전환 거리 float lod0Distance = 30.0f; // 최고 품질 float lod1Distance = 80.0f; float lod2Distance = 150.0f; float cullDistance = 500.0f; // 완전 숨김 // 이동 속도에 따른 보정 float velocityMultiplier = 1.5f; // 이동 방향으로 반경 확장 float maxVelocityBonus = 100.0f; // 최대 추가 반경 }; // 히스테리시스 적용 청크 상태 관리 class ChunkStateManager { private: enum class ChunkState { Unloaded, // 메모리에 없음 Loading, // 로딩 중 LoadedLowDetail, // 저해상도로 로딩됨 LoadedHighDetail, // 고해상도로 로딩됨 Unloading // 언로딩 중 }; std::unordered_map<ChunkCoord, ChunkState> m_chunkStates; StreamingConfig m_config; public: void UpdateChunks(const Vector3& playerPos, const Vector3& velocity) { // 이동 방향으로 반경 확장 float velocityBonus = std::min( velocity.Length() * m_config.velocityMultiplier, m_config.maxVelocityBonus ); Vector3 velocityDir = velocity.Normalized(); // 영향 범위 내 모든 청크 확인 float checkRadius = m_config.unloadRadius + velocityBonus; std::vector<ChunkCoord> chunksInRange = GetChunksInRadius(playerPos, checkRadius); for (const auto& coord : chunksInRange) { float distance = GetChunkDistance(coord, playerPos); // 이동 방향 청크는 보너스 적용 Vector3 chunkDir = (GetChunkCenter(coord) - playerPos).Normalized(); float directionBonus = std::max(0.0f, Dot(chunkDir, velocityDir)) * velocityBonus; float effectiveLoadRadius = m_config.loadRadius + directionBonus; ChunkState currentState = m_chunkStates[coord]; // 상태 전이 결정 if (distance < effectiveLoadRadius) { if (currentState == ChunkState::Unloaded) { RequestLoad(coord); } } else if (distance > m_config.unloadRadius) { if (currentState == ChunkState::LoadedHighDetail || currentState == ChunkState::LoadedLowDetail) { RequestUnload(coord); } } // loadRadius ~ unloadRadius 사이는 현재 상태 유지 (히스테리시스) } } };

Part 5. LOD(Level of Detail)와 연동

스트리밍과 LOD는 어떻게 같이 동작해?

거리에 따른 3단계 품질 관리:

가까운 거리: 고해상도 메시 + 풀 텍스처 (스트리밍 로드)

중간 거리: 중해상도 메시 + 압축 텍스처

먼 거리: 임포스터(Impostor) 또는 빌보드

Part 7. 스트리밍 스케줄러와 메모리 관리

여러 청크가 동시에 로딩 요청되면 어떻게 해?

우선순위 기반 스케줄러가 관리합니다:

• 동시 로딩 제한 (I/O 병목 방지)

• 우선순위에 따른 로딩 순서

• 메모리 예산 내 관리

// 스트리밍 스케줄러 class StreamingScheduler { private: // 로딩 대기열 (우선순위 큐) std::priority_queue<StreamingRequest> m_pendingRequests; // 현재 진행 중인 로딩 std::vector<ActiveStreamingTask> m_activeTasks; // 제한 int m_maxConcurrentLoads = 4; // 동시 로딩 최대 수 size_t m_memoryBudget = 2 * 1024 * 1024 * 1024; // 2GB size_t m_currentMemoryUsage = 0; // I/O 대역폭 제한 RateLimiter m_ioRateLimiter; // MB/s 제한 public: void RequestLoad(const StreamingRequest& request) { // 이미 로딩 중이거나 로딩됨이면 무시 if (IsAlreadyLoadedOrLoading(request.chunkId)) { return; } // 메모리 예산 확인 if (m_currentMemoryUsage + request.estimatedSize > m_memoryBudget) { // 메모리 부족 → 우선순위 낮은 청크 언로드 FreeMemoryForRequest(request.estimatedSize); } m_pendingRequests.push(request); } void Update() { // 완료된 작업 처리 ProcessCompletedTasks(); // 새 작업 시작 (동시 로딩 제한 내에서) while (m_activeTasks.size() < m_maxConcurrentLoads && !m_pendingRequests.empty()) { auto request = m_pendingRequests.top(); m_pendingRequests.pop(); // I/O 대역폭 확인 if (!m_ioRateLimiter.CanProceed(request.estimatedSize)) { m_pendingRequests.push(request); // 다시 대기열에 break; } // 로딩 시작 ActiveStreamingTask task; task.request = request; task.handle = AsyncIO::ReadAsync(request.filePath); task.startTime = Time::Now(); m_activeTasks.push_back(task); } } private: void FreeMemoryForRequest(size_t requiredSize) { // 언로드 후보 목록 (우선순위 낮은 순) auto candidates = GetUnloadCandidates(); size_t freedSize = 0; for (const auto& chunk : candidates) { if (freedSize >= requiredSize) break; UnloadChunk(chunk.id); freedSize += chunk.memorySize; m_currentMemoryUsage -= chunk.memorySize; } } std::vector<ChunkInfo> GetUnloadCandidates() { std::vector<ChunkInfo> candidates; for (const auto& chunk : m_loadedChunks) { // 플레이어와의 거리 float distance = GetDistanceToPlayer(chunk); // 마지막 접근 시간 float timeSinceAccess = Time::Now() - chunk.lastAccessTime; // 언로드 점수 계산 (높을수록 언로드 우선) float unloadScore = distance * 0.5f + timeSinceAccess * 0.3f; // 필수 청크는 언로드 불가 if (!chunk.isEssential) { candidates.push_back({chunk, unloadScore}); } } // 점수 높은 순 정렬 std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) { return a.score > b.score; }); return candidates; } };

// Unity 월드 스트리밍 매니저 using UnityEngine; using UnityEngine.AddressableAssets; using UnityEngine.ResourceManagement.AsyncOperations; using System.Collections.Generic; public class WorldStreamingManager : MonoBehaviour { [Header("Streaming Settings")] [SerializeField] private float loadRadius = 100f; [SerializeField] private float unloadRadius = 200f; [SerializeField] private float chunkSize = 50f; [SerializeField] private Transform player; private Dictionary<Vector2Int, ChunkData> loadedChunks = new Dictionary<Vector2Int, ChunkData>(); private HashSet<Vector2Int> loadingChunks = new HashSet<Vector2Int>(); private class ChunkData { public AsyncOperationHandle<GameObject> handle; public GameObject instance; } void Update() { Vector3 playerPos = player.position; Vector2Int playerChunk = WorldToChunk(playerPos); // 로딩 반경 내 청크 확인 int chunkRadius = Mathf.CeilToInt(loadRadius / chunkSize) + 1; for (int x = -chunkRadius; x <= chunkRadius; x++) { for (int z = -chunkRadius; z <= chunkRadius; z++) { Vector2Int coord = playerChunk + new Vector2Int(x, z); float distance = GetChunkDistance(coord, playerPos); if (distance < loadRadius) { if (!loadedChunks.ContainsKey(coord) && !loadingChunks.Contains(coord)) { LoadChunk(coord); } } } } // 언로딩 반경 밖 청크 해제 List<Vector2Int> toUnload = new List<Vector2Int>(); foreach (var kvp in loadedChunks) { float distance = GetChunkDistance(kvp.Key, playerPos); if (distance > unloadRadius) { toUnload.Add(kvp.Key); } } foreach (var coord in toUnload) { UnloadChunk(coord); } } private async void LoadChunk(Vector2Int coord) { loadingChunks.Add(coord); string address = $"Chunk_{coord.x}_{coord.y}"; var handle = Addressables.LoadAssetAsync<GameObject>(address); await handle.Task; if (handle.Status == AsyncOperationStatus.Succeeded) { Vector3 worldPos = ChunkToWorld(coord); GameObject instance = Instantiate(handle.Result, worldPos, Quaternion.identity); loadedChunks[coord] = new ChunkData { handle = handle, instance = instance }; } loadingChunks.Remove(coord); } private void UnloadChunk(Vector2Int coord) { if (loadedChunks.TryGetValue(coord, out ChunkData data)) { if (data.instance != null) { Destroy(data.instance); } Addressables.Release(data.handle); loadedChunks.Remove(coord); } } private Vector2Int WorldToChunk(Vector3 worldPos) { return new Vector2Int( Mathf.FloorToInt(worldPos.x / chunkSize), Mathf.FloorToInt(worldPos.z / chunkSize) ); } private Vector3 ChunkToWorld(Vector2Int coord) { return new Vector3( coord.x * chunkSize + chunkSize / 2f, 0f, coord.y * chunkSize + chunkSize / 2f ); } private float GetChunkDistance(Vector2Int coord, Vector3 playerPos) { Vector3 chunkCenter = ChunkToWorld(coord); return Vector3.Distance( new Vector3(chunkCenter.x, 0, chunkCenter.z), new Vector3(playerPos.x, 0, playerPos.z) ); } }

월드 스트리밍 최적화 10계명:

1. 히스테리시스 필수 (로드 반경 < 언로드 반경)

2. 이동 방향 예측 로딩

3. LOD와 스트리밍 분리 관리

4. 동시 로딩 수 제한 (4~8개)

5. 프레임당 GPU 업로드 제한

6. 메모리 예산 하드 리밋

7. 필수 에셋 항상 로딩 유지

8. 청크 크기 적정 선택

9. 압축 사용 (디스크/전송)

10. 시각화 디버그 모드 구현

핵심 정리

  • 월드 스트리밍 = 필요한 것만, 필요할 때만 메모리에
  • 청크(Chunk) 분할로 독립적 로드/언로드 가능하게
  • 히스테리시스: 로드 반경 < 언로드 반경 (스래싱 방지)
  • 예측 로딩: 속도, 경로, 목적지, 시야 방향
  • LOD 연동: 거리별 품질 관리로 메모리 효율화
  • 심리스 전환: 시야 차단, 포털, 안개 마스킹
  • Unity: Addressables + Scene, Unreal: World Partition
