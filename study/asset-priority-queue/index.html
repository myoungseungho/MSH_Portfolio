// 우선순위 계산 함수 class PriorityCalculator { public: LoadPriority Calculate(const AssetRequest& request, const PlayerContext& player) { float score = 0.0f; // 1. 에셋 타입 점수 (40%) score += GetTypeScore(request.AssetType) * 0.4f; // 2. 거리 점수 (30%) float dist = Vector3::Distance(player.Position, request.Position); score += GetDistanceScore(dist) * 0.3f; // 3. 가시성 점수 (20%) score += GetVisibilityScore(request.Position, player.Camera) * 0.2f; // 4. 특수 상황 점수 (10%) score += GetContextScore(request, player) * 0.1f; // 점수 → 우선순위 변환 return ScoreToPriority(score); } private: float GetDistanceScore(float distance) { if (distance < 5.0f) return 100.0f; if (distance < 15.0f) return 70.0f; if (distance < 50.0f) return 40.0f; return 10.0f; } float GetVisibilityScore(const Vector3& pos, const Camera& cam) { Vector3 toTarget = (pos - cam.Position).Normalized(); float dot = Vector3::Dot(cam.Forward, toTarget); // 시야각 내 여부 (cos(FOV/2) 기준) if (dot > 0.95f) return 100.0f; // 정중앙 if (dot > 0.7f) return 60.0f; // 화면 내 if (dot > 0.0f) return 30.0f; // 시야각 내 return 0.0f; // 뒤에 있음 } LoadPriority ScoreToPriority(float score) { if (score >= 90.0f) return LoadPriority::Critical; if (score >= 70.0f) return LoadPriority::High; if (score >= 50.0f) return LoadPriority::Medium; if (score >= 30.0f) return LoadPriority::Low; return LoadPriority::Background; } };

Part 3. 힙 자료구조와 우선순위 큐

우선순위 큐는 어떻게 구현해?

Binary Heap을 사용해!

삽입 O(log n), 최고 우선순위 추출 O(log n)으로 효율적.

Min-Heap 구조 (우선순위 값이 작을수록 높은 우선순위)

                    [0] Critical
                   /           \
             [1] High        [2] High
            /       \        /       \
       [3] Medium [4] Low [5] Medium [6] Low


배열로 표현:
┌─────┬──────┬──────┬──────┬─────┬──────┬─────┐
│  0  │  1   │  2   │  3   │  4  │  5   │  6  │
├─────┼──────┼──────┼──────┼─────┼──────┼─────┤
│Crit │High  │High  │Med   │Low  │Med   │Low  │
└─────┴──────┴──────┴──────┴─────┴──────┴─────┘

인덱스 관계:
• 부모: (i - 1) / 2
• 왼쪽 자식: 2*i + 1
• 오른쪽 자식: 2*i + 2


힙 연산:
━━━━━━━━

Push (삽입):
1. 배열 끝에 추가
2. 부모와 비교하며 위로 이동 (Bubble Up)

Pop (추출):
1. 루트(최소값) 반환
2. 마지막 요소를 루트로 이동
3. 자식과 비교하며 아래로 이동 (Bubble Down)
// 로딩 요청 구조체 struct LoadRequest { AssetKey Key; LoadPriority Priority; float Score; // 세부 정렬용 uint64_t RequestTime; // 요청 시간 (같은 우선순위 시 FIFO) Vector3 Position; // 동적 우선순위 계산용 // 비교 연산자 (Min-Heap용) bool operator>(const LoadRequest& other) const { if (Priority != other.Priority) return Priority > other.Priority; // 낮은 값이 높은 우선순위 if (std::abs(Score - other.Score) > 0.01f) return Score < other.Score; // 높은 점수가 높은 우선순위 return RequestTime > other.RequestTime; // 먼저 요청한 것 먼저 } }; // 우선순위 큐 구현 class LoadPriorityQueue { private: std::vector<LoadRequest> m_Heap; std::unordered_map<AssetKey, size_t> m_IndexMap; // 키 → 인덱스 std::mutex m_Mutex; public: void Push(const LoadRequest& request) { std::lock_guard<std::mutex> lock(m_Mutex); // 이미 있으면 우선순위 업데이트 auto it = m_IndexMap.find(request.Key); if (it != m_IndexMap.end()) { UpdatePriority(it->second, request); return; } // 새 요청 추가 size_t index = m_Heap.size(); m_Heap.push_back(request); m_IndexMap[request.Key] = index; BubbleUp(index); } bool Pop(LoadRequest& outRequest) { std::lock_guard<std::mutex> lock(m_Mutex); if (m_Heap.empty()) return false; outRequest = m_Heap[0]; m_IndexMap.erase(outRequest.Key); if (m_Heap.size() > 1) { m_Heap[0] = m_Heap.back(); m_IndexMap[m_Heap[0].Key] = 0; m_Heap.pop_back(); BubbleDown(0); } else { m_Heap.pop_back(); } return true; } private: void BubbleUp(size_t index) { while (index > 0) { size_t parent = (index - 1) / 2; if (!(m_Heap[index] > m_Heap[parent])) break; Swap(index, parent); index = parent; } } void BubbleDown(size_t index) { size_t size = m_Heap.size(); while (true) { size_t smallest = index; size_t left = 2 * index + 1; size_t right = 2 * index + 2; if (left < size && m_Heap[smallest] > m_Heap[left]) smallest = left; if (right < size && m_Heap[smallest] > m_Heap[right]) smallest = right; if (smallest == index) break; Swap(index, smallest); index = smallest; } } void Swap(size_t i, size_t j) { std::swap(m_Heap[i], m_Heap[j]); m_IndexMap[m_Heap[i].Key] = i; m_IndexMap[m_Heap[j].Key] = j; } };

Part 4. 동적 우선순위 변경

플레이어가 움직이면 우선순위가 바뀌어야 하지 않아?

맞아! 실시간 재계산이 필요해.

하지만 매 프레임 전체 재정렬은 비효율적이야.

동적 우선순위 변경 전략

[전략 1: 전체 재정렬] ❌ 비효율적
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

매 프레임:
  모든 요소 우선순위 재계산 → O(n)
  전체 힙 재구성 → O(n log n)

  요소 1000개, 60fps = 초당 60,000번 계산!


[전략 2: Lazy Update] ✅ 효율적
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• 힙에는 "예상 우선순위"만 저장
• Pop할 때 실제 우선순위 재계산
• 계산된 우선순위가 다르면?
  → 힙에 다시 넣고 다음 것 Pop

장점:
• 실제로 로딩할 것만 계산
• 대부분의 요소는 계산 생략


[전략 3: 주기적 Update] ✅ 균형 잡힌
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• 0.5초마다 우선순위 재계산
• 변화가 큰 것만 힙 조정

장점:
• 예측 가능한 성능
• 급격한 변화도 반영
// Lazy Update 방식 구현 class LazyPriorityQueue { private: LoadPriorityQueue m_Heap; PriorityCalculator m_Calculator; public: bool PopHighestPriority(LoadRequest& outRequest, const PlayerContext& player) { while (m_Heap.Pop(outRequest)) { // 실제 우선순위 재계산 LoadPriority actualPriority = m_Calculator.Calculate(outRequest, player); // 저장된 우선순위와 같거나 더 높으면 반환 if (actualPriority <= outRequest.Priority) { outRequest.Priority = actualPriority; return true; } // 우선순위가 낮아졌으면 다시 삽입 outRequest.Priority = actualPriority; m_Heap.Push(outRequest); } return false; } }; // 주기적 Update 방식 class PeriodicUpdateQueue { private: LoadPriorityQueue m_Heap; float m_UpdateInterval = 0.5f; float m_TimeSinceUpdate = 0.0f; public: void Update(float deltaTime, const PlayerContext& player) { m_TimeSinceUpdate += deltaTime; if (m_TimeSinceUpdate >= m_UpdateInterval) { RecalculateAllPriorities(player); m_TimeSinceUpdate = 0.0f; } } void RecalculateAllPriorities(const PlayerContext& player) { std::vector<LoadRequest> requests; // 모든 요청 추출 LoadRequest req; while (m_Heap.PopDirect(req)) { req.Priority = m_Calculator.Calculate(req, player); requests.push_back(req); } // 새 우선순위로 다시 삽입 for (const auto& r : requests) { m_Heap.Push(r); } } };

Part 5. 멀티스레드 로딩과 우선순위

로딩 스레드가 여러 개면 어떻게 관리해?

Work Stealing 패턴으로 효율적 분배!

각 스레드가 자기 큐에서 작업하다가, 빈 스레드가 다른 큐에서 가져옴.

// 멀티스레드 로딩 시스템 class MultiThreadedLoadSystem { private: // 우선순위별 큐 분리 std::array<LoadPriorityQueue, 6> m_PriorityQueues; // 로딩 스레드 풀 std::vector<std::thread> m_LoaderThreads; std::atomic<bool> m_Running; // 대역폭 제한 Semaphore m_BandwidthSemaphore; public: MultiThreadedLoadSystem(int threadCount, int maxConcurrent) : m_Running(true) , m_BandwidthSemaphore(maxConcurrent) // 동시 로딩 제한 { for (int i = 0; i < threadCount; ++i) { m_LoaderThreads.emplace_back(&MultiThreadedLoadSystem::LoaderThread, this); } } void Enqueue(const LoadRequest& request) { int queueIndex = static_cast<int>(request.Priority); m_PriorityQueues[queueIndex].Push(request); } private: void LoaderThread() { while (m_Running) { LoadRequest request; // 높은 우선순위 큐부터 확인 bool found = false; for (int p = 0; p < 6; ++p) { if (m_PriorityQueues[p].Pop(request)) { found = true; break; } } if (!found) { // 할 일 없으면 잠시 대기 std::this_thread::sleep_for(std::chrono::milliseconds(1)); continue; } // 대역폭 획득 대기 m_BandwidthSemaphore.Acquire(); // 실제 로딩 수행 ExecuteLoad(request); // 대역폭 반환 m_BandwidthSemaphore.Release(); } } };

Part 6. 로딩 취소와 의존성 관리

로딩 중에 취소해야 할 때도 있지 않아?

맵 전환이나 빠른 이동 시 진행 중인 로딩 취소가 필요해!

// 취소 가능한 로딩 요청 class CancellableLoadRequest { public: AssetKey Key; LoadPriority Priority; std::atomic<bool> IsCancelled{false}; std::shared_ptr<LoadResult> Result; void Cancel() { IsCancelled.store(true); } }; // 취소 처리가 있는 로더 class CancellableLoader { public: void ExecuteLoad(CancellableLoadRequest& request) { // 1. 시작 전 취소 확인 if (request.IsCancelled.load()) { return; } // 2. 파일 오픈 FileHandle file = OpenFile(request.Key); // 3. 청크 단위 읽기 (취소 포인트) std::vector<uint8_t> buffer; while (!file.IsEOF()) { // 취소 확인 if (request.IsCancelled.load()) { file.Close(); return; // 즉시 중단 } // 청크 읽기 buffer.append(file.ReadChunk(64 * 1024)); // 64KB } // 4. 마지막 취소 확인 if (!request.IsCancelled.load()) { request.Result = CreateAsset(buffer); } } };
에셋 간에 의존성이 있으면?

텍스처 → 머티리얼 → 모델처럼 의존성 그래프가 있어.

의존성이 있으면 피의존 에셋이 먼저 로딩되어야 해.

// 의존성이 있는 로딩 요청 struct DependentLoadRequest { AssetKey Key; LoadPriority BasePriority; std::vector<AssetKey> Dependencies; // 먼저 로딩해야 하는 것들 std::atomic<int> PendingDependencies; // 대기 중인 의존성 수 }; class DependencyAwareQueue { private: std::unordered_map<AssetKey, DependentLoadRequest> m_AllRequests; LoadPriorityQueue m_ReadyQueue; // 의존성 해결된 것만 public: void Enqueue(const DependentLoadRequest& request) { m_AllRequests[request.Key] = request; if (request.Dependencies.empty()) { // 의존성 없으면 바로 Ready 큐로 m_ReadyQueue.Push(ToLoadRequest(request)); } else { // 의존성 있으면 의존 에셋들 먼저 요청 for (const auto& dep : request.Dependencies) { EnsureRequested(dep, PromotePriority(request.BasePriority)); } } } void OnLoadComplete(const AssetKey& completedKey) { // 이 에셋에 의존하던 요청들 확인 for (auto& [key, req] : m_AllRequests) { if (ContainsDependency(req, completedKey)) { int remaining = --req.PendingDependencies; if (remaining == 0) { // 모든 의존성 해결! Ready 큐로 이동 m_ReadyQueue.Push(ToLoadRequest(req)); } } } } };

Part 7. 게임 엔진에서의 실제 구현

Unity에서는 어떻게 해?

Unity의 Addressables는 기본 우선순위가 없어서 직접 구현해야 해.

// Unity Addressables + 우선순위 큐 public class PriorityAddressableLoader : MonoBehaviour { private SortedDictionary<LoadPriority, Queue<LoadRequest>> _queues; private int _maxConcurrent = 3; private int _currentLoading = 0; public void Load(string address, LoadPriority priority, Action<GameObject> onComplete) { var request = new LoadRequest(address, priority, onComplete); if (_currentLoading < _maxConcurrent) { StartLoad(request); } else { _queues[priority].Enqueue(request); } } private async void StartLoad(LoadRequest request) { _currentLoading++; var handle = Addressables.LoadAssetAsync<GameObject>(request.Address); await handle.Task; request.OnComplete?.Invoke(handle.Result); _currentLoading--; ProcessNextInQueue(); } private void ProcessNextInQueue() { // 높은 우선순위부터 확인 foreach (var kvp in _queues) { if (kvp.Value.Count > 0) { var next = kvp.Value.Dequeue(); StartLoad(next); return; } } } }

Part 8. 우선순위 큐 최적화

우선순위 큐 성능을 더 높이려면?

몇 가지 고급 최적화 기법들:

// 1. Bucket 기반 우선순위 큐 (상수 시간 삽입) class BucketPriorityQueue { private: // 우선순위 레벨별 버킷 (FIFO 큐) std::array<std::deque<LoadRequest>, 6> m_Buckets; int m_HighestNonEmpty = 5; // 빈 버킷 스킵용 public: // O(1) 삽입! void Push(const LoadRequest& request) { int bucket = static_cast<int>(request.Priority); m_Buckets[bucket].push_back(request); if (bucket < m_HighestNonEmpty) { m_HighestNonEmpty = bucket; } } // O(1) ~ O(k) 추출 (k = 버킷 수) bool Pop(LoadRequest& out) { for (int i = m_HighestNonEmpty; i < 6; ++i) { if (!m_Buckets[i].empty()) { out = m_Buckets[i].front(); m_Buckets[i].pop_front(); // 빈 버킷 스킵 최적화 while (m_HighestNonEmpty < 6 && m_Buckets[m_HighestNonEmpty].empty()) { m_HighestNonEmpty++; } return true; } } return false; } }; // 2. Lock-Free 우선순위 큐 (높은 동시성) class LockFreePriorityQueue { private: // 우선순위별 Lock-Free 큐 std::array<moodycamel::ConcurrentQueue<LoadRequest>, 6> m_Queues; public: void Push(const LoadRequest& request) { int idx = static_cast<int>(request.Priority); m_Queues[idx].enqueue(request); } bool Pop(LoadRequest& out) { for (int i = 0; i < 6; ++i) { if (m_Queues[i].try_dequeue(out)) { return true; } } return false; } };

게임에서의 추천:
━━━━━━━━━━━━━━━

• 우선순위 레벨이 적으면 (5-10개): Bucket Queue
• 멀티스레드 로딩: Lock-Free Bucket Queue
• 세밀한 우선순위 필요: Binary Heap

핵심 정리

  • 플레이어가 보는 것 먼저 - 체감 로딩 시간 최소화
  • 우선순위 = 타입 + 거리 + 가시성 + 특수 상황
  • Binary Heap으로 효율적인 우선순위 큐 구현
  • 동적 우선순위 - Lazy Update 또는 주기적 재계산
  • 멀티스레드 - 우선순위별 큐 분리 + 대역폭 제한
  • 취소와 의존성 - 청크 단위 취소 포인트, 의존성 그래프
  • 최적화 - Bucket Queue로 O(1) 삽입/추출
