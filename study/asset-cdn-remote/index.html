CDN과 리모트 에셋

전 세계 유저에게 빠르고 안정적인 에셋 배포

Part 1. CDN과 리모트 에셋이란?

CDN이 뭐야?

CDN (Content Delivery Network)은 전 세계에 분산된 서버 네트워크로, 유저와 가까운 위치에서 콘텐츠를 제공하여 다운로드 속도를 높이는 시스템입니다.

게임에서 CDN이 없다면:

  • 한국 서버에서 미국 유저가 다운로드 → 높은 레이턴시
  • 대규모 패치 시 서버 과부하 → 다운로드 실패
  • 단일 서버 장애 시 → 전체 서비스 중단
리모트 에셋이 뭐야?

리모트 에셋은 앱에 포함되지 않고, 런타임에 서버(CDN)에서 다운로드하는 에셋입니다.

캐시 히트/미스가 뭐야?

요청한 콘텐츠가 Edge 서버에 있으면 캐시 히트, 없으면 캐시 미스입니다.

                                       ▼

                  + Edge/Mid-tier에 캐시됨 (다음 요청은 빠름)

캐시 히트율 높이는 방법

  • 에셋 URL에 버전 쿼리 파라미터 사용: ?v=1.2.3
  • 자주 사용되는 에셋은 미리 Edge에 "워밍업"
  • 지역별로 인기 콘텐츠 분석하여 프리캐시
  • 작은 파일은 합쳐서 요청 횟수 감소

Part 3. 다운로드 시스템 구현

리모트 에셋 다운로드 시스템은 어떻게 구현해?

견고한 다운로드 시스템에 필요한 핵심 기능들입니다:

// RemoteAssetDownloader.cs - 리모트 에셋 다운로드 시스템 using System; using System.Collections.Generic; using System.IO; using System.Threading.Tasks; using UnityEngine; using UnityEngine.Networking; public class RemoteAssetDownloader : MonoBehaviour { private const int MAX_RETRY = 3; private const int RETRY_DELAY_MS = 1000; private const int TIMEOUT_SECONDS = 30; private const int MAX_CONCURRENT = 4; private Queue<DownloadTask> m_PendingQueue = new(); private List<DownloadTask> m_ActiveDownloads = new(); private string m_CdnBaseUrl; private string m_LocalCachePath; public class DownloadTask { public string BundleName; public string Version; public long ExpectedSize; public uint ExpectedCRC; public int Priority; public int RetryCount; public Action<float> OnProgress; public Action<bool, string> OnComplete; } public void Initialize(string cdnUrl) { m_CdnBaseUrl = cdnUrl; m_LocalCachePath = Path.Combine(Application.persistentDataPath, "BundleCache"); Directory.CreateDirectory(m_LocalCachePath); } // 다운로드 요청 (우선순위 큐에 추가) public void RequestDownload(DownloadTask task) { // 이미 캐시에 있고 버전이 같으면 스킵 if (IsCached(task.BundleName, task.Version)) { task.OnComplete?.Invoke(true, "Already cached"); return; } // 우선순위 순으로 큐에 삽입 InsertByPriority(task); TryStartNextDownload(); } private void InsertByPriority(DownloadTask task) { var tempList = new List<DownloadTask>(m_PendingQueue); tempList.Add(task); tempList.Sort((a, b) => b.Priority.CompareTo(a.Priority)); m_PendingQueue = new Queue<DownloadTask>(tempList); } private void TryStartNextDownload() { while (m_ActiveDownloads.Count < MAX_CONCURRENT && m_PendingQueue.Count > 0) { var task = m_PendingQueue.Dequeue(); m_ActiveDownloads.Add(task); StartCoroutine(DownloadWithRetry(task)); } } private IEnumerator DownloadWithRetry(DownloadTask task) { string url = $"{m_CdnBaseUrl}/{task.BundleName}?v={task.Version}"; bool success = false; string error = null; while (task.RetryCount < MAX_RETRY && !success) { using (var request = UnityWebRequest.Get(url)) { request.timeout = TIMEOUT_SECONDS; // 이어받기 지원 (Range 헤더) string tempPath = GetTempPath(task.BundleName); if (File.Exists(tempPath)) { long existingSize = new FileInfo(tempPath).Length; request.SetRequestHeader("Range", $"bytes={existingSize}-"); } var handler = new DownloadHandlerFile(tempPath) { removeFileOnAbort = false }; request.downloadHandler = handler; var operation = request.SendWebRequest(); // 진행률 업데이트 while (!operation.isDone) { task.OnProgress?.Invoke(request.downloadProgress); yield return null; } if (request.result == UnityWebRequest.Result.Success) { // CRC 검증 if (VerifyCRC(tempPath, task.ExpectedCRC)) { // 임시 파일을 캐시로 이동 string cachePath = GetCachePath(task.BundleName, task.Version); File.Move(tempPath, cachePath); success = true; } else { error = "CRC mismatch"; File.Delete(tempPath); } } else { error = request.error; task.RetryCount++; if (task.RetryCount < MAX_RETRY) { yield return new WaitForSeconds(RETRY_DELAY_MS / 1000f); } } } } m_ActiveDownloads.Remove(task); task.OnComplete?.Invoke(success, success ? null : error); TryStartNextDownload(); } }

캐시 용량이 부족하면 어떻게 해?

캐시 퇴거(Eviction) 전략입니다:

// CacheManager.cs - LRU 기반 캐시 관리 public class CacheManager { private readonly long m_MaxCacheSize; private CacheIndex m_Index; private readonly string m_CachePath; public CacheManager(string cachePath, long maxSize) { m_CachePath = cachePath; m_MaxCacheSize = maxSize; m_Index = LoadIndex(); } // 새 번들 저장 전 공간 확보 public void EnsureSpace(long requiredSize) { while (m_Index.TotalSize + requiredSize > m_MaxCacheSize) { var victim = SelectEvictionCandidate(); if (victim == null) { throw new InsufficientStorageException( "필수 번들만 남아서 더 이상 삭제 불가"); } EvictBundle(victim); } } // LRU + 우선순위 기반 퇴거 후보 선정 private CacheEntry SelectEvictionCandidate() { // 우선순위 낮은 것 중에서 가장 오래된 것 선택 return m_Index.Entries.Values .Where(e => !e.IsPinned) // 고정된 번들 제외 .OrderBy(e => e.Priority) // 낮은 우선순위 먼저 .ThenBy(e => e.LastAccess) // 오래된 것 먼저 .FirstOrDefault(); } // 번들 접근 시 호출 (LRU 업데이트) public void Touch(string bundleName) { if (m_Index.Entries.TryGetValue(bundleName, out var entry)) { entry.LastAccess = DateTime.UtcNow; SaveIndex(); } } // 필수 번들 고정 (삭제 방지) public void PinBundle(string bundleName) { if (m_Index.Entries.TryGetValue(bundleName, out var entry)) { entry.IsPinned = true; SaveIndex(); } } // 스토리지 정보 조회 public CacheStats GetStats() { return new CacheStats { UsedSize = m_Index.TotalSize, MaxSize = m_MaxCacheSize, BundleCount = m_Index.Entries.Count, FreeSize = m_MaxCacheSize - m_Index.TotalSize }; } }

⚠️ 캐시 퇴거 시 주의사항

  • 현재 로드된 번들은 삭제하면 안 됨 → 언로드 후 삭제
  • 다운로드 중인 임시 파일과 구분 필요
  • 의존성 있는 번들은 함께 삭제
  • 삭제 전 유저 확인 UI 표시 고려

// NetworkAwareDownloader.cs - 네트워크 상태 감지 다운로드 public class NetworkAwareDownloader { public enum DownloadMode { Aggressive, // WiFi: 모든 에셋 프리페치 Normal, // 셀룰러: 필요한 것만 Restricted, // 저속/로밍: 필수만 Offline // 오프라인: 다운로드 중지 } private DownloadMode m_CurrentMode = DownloadMode.Normal; public void UpdateNetworkStatus() { var reachability = Application.internetReachability; switch (reachability) { case NetworkReachability.ReachableViaLocalAreaNetwork: m_CurrentMode = DownloadMode.Aggressive; break; case NetworkReachability.ReachableViaCarrierDataNetwork: // 데이터 세이버 모드 또는 로밍 체크 if (IsDataSaverEnabled() || IsRoaming()) { m_CurrentMode = DownloadMode.Restricted; } else { m_CurrentMode = DownloadMode.Normal; } break; default: m_CurrentMode = DownloadMode.Offline; break; } ApplyDownloadPolicy(); } private void ApplyDownloadPolicy() { switch (m_CurrentMode) { case DownloadMode.Aggressive: // WiFi: 백그라운드 프리페치 활성화 SetMaxConcurrent(8); EnablePrefetch(true); SetQualityPreference(AssetQuality.High); break; case DownloadMode.Normal: // 셀룰러: 필요한 것만, 적당한 품질 SetMaxConcurrent(4); EnablePrefetch(false); SetQualityPreference(AssetQuality.Medium); break; case DownloadMode.Restricted: // 제한 모드: 필수만, 저품질 SetMaxConcurrent(1); EnablePrefetch(false); SetQualityPreference(AssetQuality.Low); ShowDataWarningPopup(); break; case DownloadMode.Offline: // 오프라인: 다운로드 중지, 캐시만 사용 PauseAllDownloads(); break; } } // 셀룰러에서 대용량 다운로드 시 확인 public async Task<bool> ConfirmLargeDownload(long size) { if (m_CurrentMode == DownloadMode.Normal && size > 50 * 1024 * 1024) { // 50MB 이상 셀룰러 다운로드 시 확인 return await ShowConfirmPopup( $"셀룰러 데이터로 {FormatSize(size)}를 다운로드합니다.\n계속하시겠습니까?" ); } return true; } }

await checkHandle.Task; if (checkHandle.Status == AsyncOperationStatus.Succeeded) { return checkHandle.Result; } return new List<string>(); } // 카탈로그 업데이트 적용 public async Task<bool> UpdateCatalogs(List<string> catalogIds) { if (catalogIds.Count == 0) return true; var updateHandle = Addressables.UpdateCatalogs(catalogIds, false); await updateHandle.Task; return updateHandle.Status == AsyncOperationStatus.Succeeded; } // 다운로드 크기 확인 public async Task<long> GetDownloadSize(string label) { var sizeHandle = Addressables.GetDownloadSizeAsync(label); await sizeHandle.Task; return sizeHandle.Result; } // 라벨 기반 에셋 다운로드 public async Task DownloadAssetsByLabel(string label) { long totalSize = await GetDownloadSize(label); if (totalSize == 0) { Debug.Log($"라벨 '{label}' 에셋이 이미 캐시됨"); OnDownloadComplete?.Invoke(); return; } var downloadHandle = Addressables.DownloadDependenciesAsync(label, false); // 진행률 모니터링 while (!downloadHandle.IsDone) { var status = downloadHandle.GetDownloadStatus(); OnDownloadProgress?.Invoke(status.DownloadedBytes, status.TotalBytes); await Task.Yield(); } if (downloadHandle.Status == AsyncOperationStatus.Succeeded) { Debug.Log($"라벨 '{label}' 다운로드 완료"); OnDownloadComplete?.Invoke(); } else { Debug.LogError($"라벨 '{label}' 다운로드 실패"); } Addressables.Release(downloadHandle); } // 에셋 로드 (자동으로 캐시 또는 다운로드) public async Task<T> LoadAssetAsync<T>(string address) where T : class { var handle = Addressables.LoadAssetAsync<T>(address); await handle.Task; if (handle.Status == AsyncOperationStatus.Succeeded) { return handle.Result; } Debug.LogError($"에셋 로드 실패: {address}"); return null; } } // 사용 예시 public class GameStartup : MonoBehaviour { [SerializeField] private AddressablesRemoteLoader m_Loader; [SerializeField] private DownloadProgressUI m_ProgressUI; async void Start() { // 1. 카탈로그 업데이트 체크 var updates = await m_Loader.CheckForUpdates(); if (updates.Count > 0) { await m_Loader.UpdateCatalogs(updates); } // 2. 필수 에셋 다운로드 크기 확인 long size = await m_Loader.GetDownloadSize("essential"); if (size > 0) { m_ProgressUI.Show($"다운로드 필요: {size / 1024 / 1024}MB"); m_Loader.OnDownloadProgress = (downloaded, total) => { m_ProgressUI.UpdateProgress(downloaded, total); }; await m_Loader.DownloadAssetsByLabel("essential"); } // 3. 게임 시작 SceneManager.LoadScene("MainMenu"); } }

핵심 정리: CDN과 리모트 에셋

  • CDN: 전 세계 분산 서버로 빠른 에셋 전송
  • 리모트 에셋: 앱 크기 최소화, 스토어 심사 없이 업데이트
  • 캐시 관리: LRU 퇴거, 무결성 검증, 용량 제한
  • 네트워크: WiFi/셀룰러 구분, 병렬 다운로드, 이어받기
  • 보안: URL 서명, 암호화, 무결성 해시
  • 비용: 압축 최적화, 캐시 히트율 향상, 델타 패치
