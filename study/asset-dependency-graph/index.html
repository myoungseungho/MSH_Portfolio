에셋 의존성 그래프 (Asset Dependency Graph)

에셋 간 참조 관계를 그래프로 모델링하여 빌드, 로드, 번들링 최적화

Part 1. 개념 - 의존성 그래프란?

에셋 의존성 그래프가 정확히 뭐야?

에셋 의존성 그래프(Asset Dependency Graph)는 에셋 간의 참조 관계를 방향 그래프(Directed Graph)로 표현한 자료구조야.

그래프 용어:

  • 노드(Node/Vertex): 개별 에셋
  • 엣지(Edge): 에셋 간 참조 관계
  • 방향: A→B = "A가 B를 참조"
  • DAG: 순환이 없는 방향 그래프 (이상적)

의존성 그래프는 어떻게 표현해?

프로그래밍에서 그래프는 여러 방식으로 표현할 수 있어:

// 의존성 그래프 표현 방식 // 1. 인접 리스트 (Adjacency List) - 가장 일반적 public class DependencyGraph { // 각 노드가 참조하는 노드들의 목록 private Dictionary<string, HashSet<string>> m_Adjacency; // 예시 데이터: // "PlayerPrefab" → {"PlayerMesh", "PlayerMat"} // "PlayerMat" → {"PlayerTexture", "SharedShader"} // "SharedShader" → {} (리프 노드) } // 2. 엣지 리스트 (Edge List) public class DependencyEdge { public string From; // 참조하는 에셋 public string To; // 참조되는 에셋 } List<DependencyEdge> edges = new() { new { From = "PlayerPrefab", To = "PlayerMesh" }, new { From = "PlayerPrefab", To = "PlayerMat" }, new { From = "PlayerMat", To = "PlayerTexture" }, // ... }; // 3. 인접 행렬 (Adjacency Matrix) - 밀집 그래프용 // 에셋 수가 많으면 메모리 비효율적 (N×N) bool[,] adjacencyMatrix = new bool[assetCount, assetCount]; adjacencyMatrix[playerPrefabIndex, playerMeshIndex] = true;

표현 방식 비교

방식 공간 복잡도 엣지 조회 이웃 조회 적합한 경우
인접 리스트 O(V + E) O(degree) O(1) 희소 그래프 (일반적)
엣지 리스트 O(E) O(E) O(E) 엣지 순회 중심

Part 2. 왜 의존성 그래프가 필요해?

의존성 그래프를 왜 분석해야 해?

의존성 그래프 분석은 에셋 관리의 모든 측면에 영향을 미쳐:

의존성 그래프 활용 분야

1. 에셋 번들 최적화

  • 공유 에셋 분리로 중복 방지
  • 번들 크기 최소화
  • 로드 순서 결정

2. 빌드 최적화

  • 변경된 에셋의 영향 범위 파악
  • 인크리멘탈 빌드 최적화
  • 병렬 빌드 가능 여부 판단

3. 런타임 로딩

  • 프리로드 대상 결정
  • 언로드 안전성 검사
  • 메모리 예측

4. 품질 관리

  • 순환 참조 감지
  • 미사용 에셋 발견
  • 과도한 의존성 경고

// 의존성 기반 번들 최적화 알고리즘 public class BundleOptimizer { private DependencyGraph m_Graph; public List<BundleDefinition> OptimizeBundles( List<string> bundleRoots) { // 1. 각 번들 루트에서 도달 가능한 모든 에셋 수집 var bundleAssets = new Dictionary<string, HashSet<string>>(); foreach (var root in bundleRoots) { bundleAssets[root] = m_Graph.GetAllDependencies(root); } // 2. 공유 에셋 식별 (2개 이상 번들에서 참조) var refCount = new Dictionary<string, int>(); foreach (var kvp in bundleAssets) { foreach (var asset in kvp.Value) { refCount[asset] = refCount.GetValueOrDefault(asset, 0) + 1; } } var sharedAssets = refCount .Where(kvp => kvp.Value > 1) .Select(kvp => kvp.Key) .ToHashSet(); // 3. 공유 번들 생성 var result = new List<BundleDefinition>(); if (sharedAssets.Count > 0) { result.Add(new BundleDefinition { Name = "shared_core", Assets = sharedAssets.ToList(), Dependencies = new List<string>() }); } // 4. 개별 번들 생성 (공유 에셋 제외) foreach (var kvp in bundleAssets) { var uniqueAssets = kvp.Value.Except(sharedAssets).ToList(); result.Add(new BundleDefinition { Name = kvp.Key, Assets = uniqueAssets, Dependencies = sharedAssets.Count > 0 ? new List<string> { "shared_core" } : new List<string>() }); } return result; } }

Part 3. 그래프 알고리즘

의존성 그래프 분석에 어떤 알고리즘이 사용돼?

에셋 관리에서 자주 사용되는 그래프 알고리즘들:

// 핵심 그래프 알고리즘 public class GraphAlgorithms { // ═══ 1. DFS (깊이 우선 탐색) ═══ // 용도: 의존성 수집, 순환 감지 public HashSet<string> GetAllDependencies(string root) { var visited = new HashSet<string>(); DFS(root, visited); return visited; } private void DFS(string node, HashSet<string> visited) { if (visited.Contains(node)) return; visited.Add(node); foreach (var neighbor in GetNeighbors(node)) DFS(neighbor, visited); } // ═══ 2. BFS (너비 우선 탐색) ═══ // 용도: 최단 경로, 레벨별 의존성 public Dictionary<string, int> GetDependencyLevels(string root) { var levels = new Dictionary<string, int>(); var queue = new Queue<string>(); queue.Enqueue(root); levels[root] = 0; while (queue.Count > 0) { var current = queue.Dequeue(); var currentLevel = levels[current]; foreach (var neighbor in GetNeighbors(current)) { if (!levels.ContainsKey(neighbor)) { levels[neighbor] = currentLevel + 1; queue.Enqueue(neighbor); } } } return levels; } // ═══ 3. 위상 정렬 (Topological Sort) ═══ // 용도: 로드/빌드 순서 결정 public List<string> TopologicalSort() { var inDegree = new Dictionary<string, int>(); var result = new List<string>(); var queue = new Queue<string>(); // 진입 차수 계산 foreach (var node in AllNodes) { inDegree[node] = GetIncomingEdges(node).Count; if (inDegree[node] == 0) queue.Enqueue(node); } // Kahn's Algorithm while (queue.Count > 0) { var current = queue.Dequeue(); result.Add(current); foreach (var neighbor in GetNeighbors(current)) { inDegree[neighbor]--; if (inDegree[neighbor] == 0) queue.Enqueue(neighbor); } } // 순환 감지 if (result.Count != AllNodes.Count) throw new CyclicDependencyException("그래프에 순환이 있습니다!"); return result; } // ═══ 4. 순환 감지 (Cycle Detection) ═══ // 용도: 순환 참조 발견 public List<List<string>> FindCycles() { var cycles = new List<List<string>>(); var visited = new HashSet<string>(); var recStack = new HashSet<string>(); var path = new List<string>(); foreach (var node in AllNodes) { if (!visited.Contains(node)) DFSCycleDetect(node, visited, recStack, path, cycles); } return cycles; } private void DFSCycleDetect( string node, HashSet<string> visited, HashSet<string> recStack, List<string> path, List<List<string>> cycles) { visited.Add(node); recStack.Add(node); path.Add(node); foreach (var neighbor in GetNeighbors(node)) { if (!visited.Contains(neighbor)) { DFSCycleDetect(neighbor, visited, recStack, path, cycles); } else if (recStack.Contains(neighbor)) { // 순환 발견! var cycleStart = path.IndexOf(neighbor); var cycle = path.Skip(cycleStart).ToList(); cycle.Add(neighbor); // 순환 완성 cycles.Add(cycle); } } recStack.Remove(node); path.RemoveAt(path.Count - 1); } }

// 위상 정렬 기반 병렬 에셋 로더 public class TopologicalAssetLoader { private DependencyGraph m_Graph; public async Task LoadAssetsParallel(string rootAsset) { // 1. 의존성 수집 var allDeps = m_Graph.GetAllDependencies(rootAsset); // 2. 레벨별 그룹화 var levels = GetTopologicalLevels(allDeps); // 3. 레벨 순서대로 로드 (같은 레벨은 병렬) foreach (var level in levels) { // 같은 레벨의 에셋들은 서로 의존성이 없으므로 병렬 로드 가능 var loadTasks = level.Select(asset => LoadAssetAsync(asset)); await Task.WhenAll(loadTasks); Debug.Log($"Level {levels.IndexOf(level)} 완료: {level.Count}개 에셋"); } } private List<List<string>> GetTopologicalLevels(HashSet<string> assets) { var levels = new List<List<string>>(); var remaining = new HashSet<string>(assets); var completed = new HashSet<string>(); while (remaining.Count > 0) { // 현재 레벨: 의존성이 모두 완료된 에셋들 var currentLevel = remaining .Where(a => m_Graph.GetDependencies(a) .All(d => completed.Contains(d) || !assets.Contains(d))) .ToList(); if (currentLevel.Count == 0) throw new Exception("순환 의존성 발견!"); levels.Add(currentLevel); foreach (var asset in currentLevel) { remaining.Remove(asset); completed.Add(asset); } } return levels; } }

// Unity 의존성 그래프 분석 도구 using UnityEditor; using UnityEngine; using System.Collections.Generic; using System.Linq; public class DependencyGraphAnalyzer : EditorWindow { [MenuItem("Tools/Dependency Graph Analyzer")] public static void ShowWindow() { GetWindow<DependencyGraphAnalyzer>("Dependency Analyzer"); } private Dictionary<string, HashSet<string>> m_Graph; private Dictionary<string, HashSet<string>> m_ReverseGraph; // 전체 프로젝트 그래프 빌드 public void BuildProjectGraph() { m_Graph = new Dictionary<string, HashSet<string>>(); m_ReverseGraph = new Dictionary<string, HashSet<string>>(); var allAssets = AssetDatabase.GetAllAssetPaths() .Where(p => p.StartsWith("Assets/")) .ToList(); foreach (var asset in allAssets) { // 직접 의존성만 (재귀 X) var deps = AssetDatabase.GetDependencies(asset, false) .Where(d => d != asset) // 자기 참조 제외 .ToHashSet(); m_Graph[asset] = deps; // 역방향 그래프도 구축 foreach (var dep in deps) { if (!m_ReverseGraph.ContainsKey(dep)) m_ReverseGraph[dep] = new HashSet<string>(); m_ReverseGraph[dep].Add(asset); } } Debug.Log($"그래프 빌드 완료: {m_Graph.Count} 노드"); } // 가장 많이 참조되는 에셋 (허브 노드) public List<(string asset, int refCount)> FindHubNodes(int topN = 20) { return m_ReverseGraph .OrderByDescending(kvp => kvp.Value.Count) .Take(topN) .Select(kvp => (kvp.Key, kvp.Value.Count)) .ToList(); } // 고아 에셋 (아무도 참조 안 하고 씬에도 없음) public List<string> FindOrphanAssets() { var usedAssets = new HashSet<string>(); // 빌드 설정의 씬들에서 사용되는 에셋 수집 foreach (var scene in EditorBuildSettings.scenes) { if (!scene.enabled) continue; var deps = AssetDatabase.GetDependencies(scene.path, true); foreach (var dep in deps) usedAssets.Add(dep); } return m_Graph.Keys .Where(a => !usedAssets.Contains(a)) .Where(a => !a.EndsWith(".cs")) // 스크립트 제외 .ToList(); } // 깊은 의존성 체인 탐지 public List<(string root, int depth)> FindDeepChains(int minDepth = 8) { var result = new List<(string, int)>(); foreach (var root in m_Graph.Keys) { var maxDepth = CalculateMaxDepth(root, new HashSet<string>()); if (maxDepth >= minDepth) result.Add((root, maxDepth)); } return result.OrderByDescending(x => x.depth).ToList(); } private int CalculateMaxDepth(string node, HashSet<string> visited) { if (visited.Contains(node)) return 0; // 순환 방지 visited.Add(node); if (!m_Graph.ContainsKey(node) || m_Graph[node].Count == 0) return 0; var maxChildDepth = m_Graph[node] .Max(child => CalculateMaxDepth(child, new HashSet<string>(visited))); return maxChildDepth + 1; } }

Part 5. 문제 패턴과 해결

의존성 그래프에서 흔한 문제 패턴은?

대규모 프로젝트에서 자주 발생하는 의존성 문제들:

의존성 그래프 안티패턴

1. 순환 참조 (Circular Dependency)

  • A→B→C→A 형태의 순환
  • 빌드/로드 순서 결정 불가
  • 메모리 누수 위험

2. 거대 허브 노드 (God Asset)

  • 수백 개의 에셋에서 참조되는 단일 에셋
  • 변경 시 파급 효과 거대
  • 번들 분리 어려움

3. 깊은 체인 (Deep Chain)

  • A→B→C→D→E→F... 10단계 이상
  • 로드 시간 증가
  • 디버깅 어려움

4. 숨은 의존성 (Hidden Dependency)

  • 문자열 기반 동적 로드
  • 정적 분석 불가
  • 런타임에만 발견

허브 노드 문제는 어떻게 해결해?

허브 노드(God Asset) 문제를 해결하는 전략:

// 허브 노드 감지 및 분리 도구 public class HubNodeResolver { private const int HUB_THRESHOLD = 50; // 50개 이상 참조 시 허브 public List<HubNodeReport> AnalyzeHubNodes() { var reports = new List<HubNodeReport>(); foreach (var asset in m_ReverseGraph.Keys) { var refCount = m_ReverseGraph[asset].Count; if (refCount >= HUB_THRESHOLD) { reports.Add(new HubNodeReport { AssetPath = asset, ReferenceCount = refCount, Referrers = m_ReverseGraph[asset].ToList(), SuggestedAction = SuggestAction(asset, refCount) }); } } return reports.OrderByDescending(r => r.ReferenceCount).ToList(); } private string SuggestAction(string asset, int refCount) { var ext = Path.GetExtension(asset).ToLower(); return ext switch { // 셰이더: 의도적 공유일 수 있음 ".shader" => refCount > 200 ? "경고: 매우 많은 참조. 변형별로 분리 검토" : "정상: 셰이더는 공유가 일반적", // 텍스처: 아틀라스 분리 검토 ".png" or ".jpg" => refCount > 100 ? "제안: 텍스처 아틀라스를 용도별로 분리" : "주의: 공유 텍스처 사용 패턴 검토", // 프리팹: 문제 가능성 높음 ".prefab" => "경고: 프리팹이 과도하게 공유됨. 컴포넌트 분리 검토", // ScriptableObject: 설정 파일은 괜찮음 ".asset" => asset.Contains("Settings") ? "정상: 설정 파일의 공유는 일반적" : "검토: 역할별 SO 분리 고려", _ => "검토: 과도한 공유 패턴" }; } } // 허브 노드 분리 예시 Before: SharedAtlas.png (참조: 347개) ├── UI 요소 100개 ├── 캐릭터 아이콘 150개 └── 아이템 아이콘 97개 After: UIAtlas.png (참조: 100개) CharacterIconAtlas.png (참조: 150개) ItemIconAtlas.png (참조: 97개) 효과: - 부분 업데이트 가능 (아이템만 변경 시 아이템 아틀라스만) - 번들 분리 용이 - 메모리 필요 시점에만 로드 가능

Part 7. 실전 최적화

의존성 그래프를 활용한 빌드 최적화는?

의존성 그래프 분석으로 빌드 시간과 번들 크기를 크게 줄일 수 있어:

// 빌드 영향 분석기 public class BuildImpactAnalyzer { // 변경된 에셋의 영향 범위 분석 public BuildImpactReport AnalyzeImpact(List<string> changedAssets) { var report = new BuildImpactReport(); foreach (var asset in changedAssets) { // 직접 영향 (이 에셋을 참조하는 것들) var directImpact = GetReferencers(asset); report.DirectlyAffected.UnionWith(directImpact); // 간접 영향 (전이적 참조) var indirectImpact = new HashSet<string>(); foreach (var direct in directImpact) { indirectImpact.UnionWith(GetReferencers(direct)); } report.IndirectlyAffected.UnionWith(indirectImpact); // 영향받는 번들 var affectedBundles = directImpact .Concat(indirectImpact) .Select(a => GetContainingBundle(a)) .Where(b => b != null) .Distinct(); report.AffectedBundles.UnionWith(affectedBundles); } // 재빌드 필요 범위 계산 report.RebuildRequired = report.DirectlyAffected .Concat(report.IndirectlyAffected) .Distinct() .ToList(); return report; } } // 사용 예시 var changedFiles = new List<string> { "Assets/Shaders/StandardLit.shader" }; var impact = analyzer.AnalyzeImpact(changedFiles); Debug.Log($"변경된 에셋: {changedFiles.Count}개"); Debug.Log($"직접 영향: {impact.DirectlyAffected.Count}개"); Debug.Log($"간접 영향: {impact.IndirectlyAffected.Count}개"); Debug.Log($"재빌드 필요 번들: {impact.AffectedBundles.Count}개"); // 출력 예시: // 변경된 에셋: 1개 // 직접 영향: 847개 (이 셰이더를 쓰는 머티리얼) // 간접 영향: 2,341개 (그 머티리얼을 쓰는 프리팹/씬) // 재빌드 필요 번들: 23개
런타임 메모리 예측에 어떻게 활용해?

의존성 그래프로 씬/번들 로드 시 필요한 메모리를 예측할 수 있어:

// 메모리 예측 시스템 public class MemoryPredictor { private Dictionary<string, long> m_AssetSizes; public MemoryPrediction PredictMemoryUsage(string sceneOrBundle) { var prediction = new MemoryPrediction(); // 전체 의존성 수집 var allDeps = GetAllDependencies(sceneOrBundle); foreach (var asset in allDeps) { var size = GetRuntimeSize(asset); var category = CategorizeAsset(asset); prediction.TotalSize += size; prediction.ByCategory[category] += size; prediction.Assets.Add(new AssetMemory { Path = asset, Size = size }); } // 공유 에셋 분석 var currentlyLoaded = GetCurrentlyLoadedAssets(); var shared = allDeps.Intersect(currentlyLoaded); prediction.SharedSize = shared.Sum(a => GetRuntimeSize(a)); prediction.NewLoadSize = prediction.TotalSize - prediction.SharedSize; return prediction; } private long GetRuntimeSize(string assetPath) { // 에셋 타입에 따른 런타임 메모리 추정 var ext = Path.GetExtension(assetPath).ToLower(); var fileSize = new FileInfo(assetPath).Length; return ext switch { // 텍스처: 압축 해제 크기 예측 ".png" or ".jpg" => EstimateTextureRuntimeSize(assetPath), // 메시: 버텍스/인덱스 버퍼 ".fbx" or ".obj" => fileSize * 2, // 대략적 추정 // 오디오: 압축 방식에 따라 ".wav" => fileSize, // 비압축 ".ogg" => fileSize * 10, // 스트리밍 시 버퍼 // 기타 _ => fileSize }; } } // 사용 예시 var prediction = predictor.PredictMemoryUsage("Assets/Scenes/BossRoom.unity"); Debug.Log($"=== 메모리 예측: BossRoom ==="); Debug.Log($"총 필요 메모리: {prediction.TotalSize / 1024f / 1024f:F1} MB"); Debug.Log($"이미 로드됨: {prediction.SharedSize / 1024f / 1024f:F1} MB"); Debug.Log($"새로 로드: {prediction.NewLoadSize / 1024f / 1024f:F1} MB"); Debug.Log($"\n카테고리별:"); foreach (var kvp in prediction.ByCategory) Debug.Log($" {kvp.Key}: {kvp.Value / 1024f / 1024f:F1} MB"); // 출력 예시: // === 메모리 예측: BossRoom === // 총 필요 메모리: 847.3 MB // 이미 로드됨: 234.5 MB (공유 에셋) // 새로 로드: 612.8 MB // // 카테고리별: // Texture: 523.4 MB // Mesh: 189.2 MB // Audio: 98.7 MB // Animation: 36.0 MB

핵심 요약: 에셋 의존성 그래프

  • 정의: 에셋 간 참조 관계를 방향 그래프로 표현
  • 표현: 인접 리스트 (희소 그래프에 적합)
  • 핵심 알고리즘: DFS (의존성 수집), BFS (레벨), 위상 정렬 (로드 순서)
