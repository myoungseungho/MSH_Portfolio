로딩 속도 vs 파일 크기

압축과 해제의 트레이드오프, 플랫폼별 최적화 전략

Part 1. 개념: 트레이드오프의 본질

로딩 속도와 파일 크기 사이에 왜 트레이드오프가 있어?

파일을 압축하면 크기는 줄어들지만, 로딩할 때 해제(Decompression) 비용이 발생해. 반대로 압축하지 않으면 파일은 크지만 바로 사용할 수 있어.

이 트레이드오프의 핵심은 I/O 속도 vs CPU 속도야:

다운로드와 런타임 로딩에 다른 압축을 쓸 수 있어?

이중 압축 전략이 바로 그거야. 다운로드 크기는 최소화하고, 로딩 속도는 최대화하는 방법:

// 이중 압축 전략 구현 public class DualCompressionManager { private readonly string _downloadCachePath; private readonly string _runtimeCachePath; /// <summary> /// CDN에서 LZMA 압축 번들 다운로드 후 LZ4로 재압축 /// </summary> public async Task DownloadAndRecompressAsync(string bundleUrl) { var bundleName = Path.GetFileName(bundleUrl); var downloadPath = Path.Combine(_downloadCachePath, bundleName); var runtimePath = Path.Combine(_runtimeCachePath, bundleName); // 1. LZMA 압축 번들 다운로드 (작은 크기) await DownloadFileAsync(bundleUrl, downloadPath); // 2. LZMA 해제 + LZ4 재압축 (백그라운드) await Task.Run(() => { // LZMA 번들 로드 (전체 해제됨) var bundle = AssetBundle.LoadFromFile(downloadPath); // LZ4로 재압축하여 저장 RecompressAsLZ4(bundle, runtimePath); bundle.Unload(true); }); // 3. LZMA 원본 삭제 File.Delete(downloadPath); Debug.Log($"Bundle recompressed: {bundleName}"); } /// <summary> /// 런타임에는 LZ4 번들 사용 (빠른 로딩) /// </summary> public AssetBundle LoadBundle(string bundleName) { var runtimePath = Path.Combine(_runtimeCachePath, bundleName); // LZ4 청크 기반 로딩 - 필요한 에셋만 해제 return AssetBundle.LoadFromFile(runtimePath); } } // 효과 예시: // - 다운로드: 100MB (LZMA) vs 150MB (LZ4) → 50MB 절약 // - 로딩: 0.2s (LZ4) vs 1.5s (LZMA) → 7.5배 빠름

Part 3. 실측 벤치마크와 분석

실제로 어떻게 벤치마크해?

정확한 최적화를 위해서는 타겟 디바이스에서 실측해야 해:

// 로딩 성능 벤치마크 시스템 public class LoadingBenchmark { public struct BenchmarkResult { public string BundleName; public long FileSizeBytes; public float LoadTimeMs; public float DecompressTimeMs; public float TotalTimeMs; public float PeakMemoryMB; public float EffectiveThroughputMBps; } public async Task<BenchmarkResult> RunBenchmarkAsync(string bundlePath) { var result = new BenchmarkResult(); result.BundleName = Path.GetFileName(bundlePath); result.FileSizeBytes = new FileInfo(bundlePath).Length; // GC 정리 후 시작 GC.Collect(); GC.WaitForPendingFinalizers(); var startMemory = Profiler.GetTotalAllocatedMemoryLong(); // 디스크 캐시 무효화 (정확한 I/O 측정) await InvalidateDiskCache(bundlePath); // 실제 로딩 측정 var sw = Stopwatch.StartNew(); var request = AssetBundle.LoadFromFileAsync(bundlePath); while (!request.isDone) { await Task.Yield(); } sw.Stop(); result.TotalTimeMs = sw.ElapsedMilliseconds; // 메모리 측정 var endMemory = Profiler.GetTotalAllocatedMemoryLong(); result.PeakMemoryMB = (endMemory - startMemory) / (1024f * 1024f); // 처리량 계산 result.EffectiveThroughputMBps = (result.FileSizeBytes / (1024f * 1024f)) / (result.TotalTimeMs / 1000f); // 정리 request.assetBundle?.Unload(true); return result; } public void PrintResults(List<BenchmarkResult> results) { var sb = new StringBuilder(); sb.AppendLine("=== Loading Benchmark Results ==="); sb.AppendLine($"Device: {SystemInfo.deviceModel}"); sb.AppendLine($"Platform: {Application.platform}"); sb.AppendLine(); foreach (var r in results) { sb.AppendLine($"Bundle: {r.BundleName}"); sb.AppendLine($" Size: {r.FileSizeBytes / 1024f / 1024f:F2} MB"); sb.AppendLine($" Time: {r.TotalTimeMs:F1} ms"); sb.AppendLine($" Throughput: {r.EffectiveThroughputMBps:F1} MB/s"); sb.AppendLine($" Peak Memory: {r.PeakMemoryMB:F1} MB"); sb.AppendLine(); } Debug.Log(sb.ToString()); } }

Part 4. 플랫폼별 최적화 전략

플랫폼별로 어떤 전략을 써야 해?

타겟 플랫폼의 특성에 맞춰 압축 전략을 선택해:

// 플랫폼별 압축 설정 관리 public class PlatformCompressionConfig { public static CompressionSettings GetOptimalSettings() { var platform = Application.platform; var deviceTier = GetDeviceTier(); return platform switch { // 모바일 RuntimePlatform.Android => GetAndroidSettings(deviceTier), RuntimePlatform.IPhonePlayer => GetIOSSettings(deviceTier), // PC RuntimePlatform.WindowsPlayer => GetPCSettings(), RuntimePlatform.OSXPlayer => GetPCSettings(), RuntimePlatform.LinuxPlayer => GetPCSettings(), // 콘솔 RuntimePlatform.PS5 => GetPS5Settings(), RuntimePlatform.XboxSeriesX => GetXboxSettings(), RuntimePlatform.Switch => GetSwitchSettings(), _ => GetDefaultSettings() }; } private static CompressionSettings GetAndroidSettings(DeviceTier tier) { return tier switch { // 저사양: eMMC 느림, CPU 약함 → 중간 압축 DeviceTier.Low => new CompressionSettings { BundleCompression = CompressionType.LZ4, TextureCompression = TextureCompressionFormat.ETC2, AudioCompression = AudioCompressionFormat.Vorbis, AudioQuality = 0.5f, UseAsyncLoading = true, MaxConcurrentLoads = 2 // I/O 제한 }, // 중간: UFS 3.0, 적당한 CPU → LZ4 최적 DeviceTier.Medium => new CompressionSettings { BundleCompression = CompressionType.LZ4, TextureCompression = TextureCompressionFormat.ASTC_6x6, AudioCompression = AudioCompressionFormat.Vorbis, AudioQuality = 0.7f, UseAsyncLoading = true, MaxConcurrentLoads = 4 }, // 고사양: UFS 3.1+, 강력한 CPU → 비압축 가능 DeviceTier.High => new CompressionSettings { BundleCompression = CompressionType.Uncompressed, TextureCompression = TextureCompressionFormat.ASTC_4x4, AudioCompression = AudioCompressionFormat.PCM, // 무압축 AudioQuality = 1.0f, UseAsyncLoading = true, MaxConcurrentLoads = 8 }, _ => GetDefaultSettings() }; } private static CompressionSettings GetPS5Settings() { // PS5: 커스텀 SSD + 하드웨어 Kraken 해제 return new CompressionSettings { // Oodle Kraken = 하드웨어 가속 해제 BundleCompression = CompressionType.Oodle, TextureCompression = TextureCompressionFormat.BC7, AudioCompression = AudioCompressionFormat.Opus, AudioQuality = 0.9f, UseAsyncLoading = true, MaxConcurrentLoads = 16 // SSD가 매우 빠름 }; } private static CompressionSettings GetSwitchSettings() { // Switch: 카트리지/SD카드 I/O 제한 + 약한 CPU return new CompressionSettings { BundleCompression = CompressionType.LZ4, // CPU 부담 최소화 TextureCompression = TextureCompressionFormat.ASTC_8x8, // 작은 크기 AudioCompression = AudioCompressionFormat.ADPCM, AudioQuality = 0.6f, UseAsyncLoading = true, MaxConcurrentLoads = 2 // I/O 병목 }; } }
디바이스 티어는 어떻게 판단해?

런타임에 디바이스 성능을 프로파일링해서 판단해:

// 디바이스 티어 판별 public class DeviceTierDetector { public static DeviceTier DetectTier() { var score = 0; // CPU 코어 수 var cpuCores = SystemInfo.processorCount; if (cpuCores >= 8) score += 30; else if (cpuCores >= 6) score += 20; else if (cpuCores >= 4) score += 10; // 메모리 var memoryGB = SystemInfo.systemMemorySize / 1024f; if (memoryGB >= 8) score += 30; else if (memoryGB >= 6) score += 20; else if (memoryGB >= 4) score += 10; // GPU var gpuMemoryMB = SystemInfo.graphicsMemorySize; if (gpuMemoryMB >= 4096) score += 20; else if (gpuMemoryMB >= 2048) score += 10; // I/O 벤치마크 (실제 측정) var ioScore = RunIOBenchmark(); score += ioScore; // 티어 결정 if (score >= 70) return DeviceTier.High; if (score >= 40) return DeviceTier.Medium; return DeviceTier.Low; } private static int RunIOBenchmark() { // 작은 파일 읽기 테스트로 I/O 속도 측정 var testFile = Path.Combine(Application.temporaryCachePath, "io_test.bin"); // 1MB 테스트 파일 생성 var testData = new byte[1024 * 1024]; File.WriteAllBytes(testFile, testData); // 읽기 속도 측정 var sw = Stopwatch.StartNew(); for (int i = 0; i < 10; i++) { File.ReadAllBytes(testFile); } sw.Stop(); File.Delete(testFile); // MB/s 계산 var mbps = 10f / (sw.ElapsedMilliseconds / 1000f); // I/O 점수 (UFS 3.1 = ~1500 MB/s, eMMC = ~200 MB/s) if (mbps >= 800) return 20; if (mbps >= 400) return 10; return 0; } } public enum DeviceTier { Low, Medium, High }

// 텍스처 타입별 압축 설정 public class TextureCompressionRules : AssetPostprocessor { void OnPreprocessTexture() { var importer = assetImporter as TextureImporter; var path = assetPath.ToLower(); // 텍스처 타입 판별 if (path.Contains("/ui/") || path.Contains("/sprites/")) { ConfigureUITexture(importer); } else if (path.Contains("/normalmap/") || path.EndsWith("_n.png")) { ConfigureNormalMap(importer); } else if (path.Contains("/lightmap/")) { ConfigureLightmap(importer); } else if (path.Contains("/hdr/") || path.EndsWith(".exr")) { ConfigureHDRTexture(importer); } else { ConfigureDefaultTexture(importer); } } private void ConfigureUITexture(TextureImporter importer) { // UI: 선명해야 함, 압축 최소화 importer.textureType = TextureImporterType.Sprite; importer.mipmapEnabled = false; // UI는 밉맵 불필요 importer.filterMode = FilterMode.Bilinear; var androidSettings = importer.GetPlatformTextureSettings("Android"); androidSettings.overridden = true; androidSettings.format = TextureImporterFormat.ASTC_4x4; // 고품질 androidSettings.compressionQuality = 100; importer.SetPlatformTextureSettings(androidSettings); var iosSettings = importer.GetPlatformTextureSettings("iPhone"); iosSettings.overridden = true; iosSettings.format = TextureImporterFormat.ASTC_4x4; importer.SetPlatformTextureSettings(iosSettings); } private void ConfigureNormalMap(TextureImporter importer) { // 노말맵: 전용 압축, 품질 중요 importer.textureType = TextureImporterType.NormalMap; importer.mipmapEnabled = true; importer.filterMode = FilterMode.Trilinear; var androidSettings = importer.GetPlatformTextureSettings("Android"); androidSettings.overridden = true; androidSettings.format = TextureImporterFormat.ASTC_5x5; // 노말맵 밸런스 importer.SetPlatformTextureSettings(androidSettings); var pcSettings = importer.GetPlatformTextureSettings("Standalone"); pcSettings.overridden = true; pcSettings.format = TextureImporterFormat.BC5; // 2채널 노말맵 전용 importer.SetPlatformTextureSettings(pcSettings); } private void ConfigureDefaultTexture(TextureImporter importer) { // 일반 텍스처: 밸런스 importer.textureType = TextureImporterType.Default; importer.mipmapEnabled = true; importer.streamingMipmaps = true; // 밉맵 스트리밍 활성화 importer.streamingMipmapsPriority = 0; var androidSettings = importer.GetPlatformTextureSettings("Android"); androidSettings.overridden = true; androidSettings.format = TextureImporterFormat.ASTC_6x6; // 밸런스 androidSettings.maxTextureSize = 2048; importer.SetPlatformTextureSettings(androidSettings); } }

오디오도 압축 전략이 따로 있어?

오디오는 재생 방식에 따라 압축을 다르게 적용해:

│  ├─ 장점: 최소 메모리 사용                                         │
│  ├─ 단점: I/O 발생, 시작 지연                                      │

// 오디오 압축 자동 설정 public class AudioCompressionRules : AssetPostprocessor { void OnPreprocessAudio() { var importer = assetImporter as AudioImporter; var path = assetPath.ToLower(); // 오디오 타입 판별 if (path.Contains("/sfx/")) { ConfigureSFX(importer); } else if (path.Contains("/bgm/") || path.Contains("/music/")) { ConfigureBGM(importer); } else if (path.Contains("/voice/") || path.Contains("/dialogue/")) { ConfigureVoice(importer); } else if (path.Contains("/ambient/")) { ConfigureAmbient(importer); } } private void ConfigureSFX(AudioImporter importer) { // SFX: 즉시 재생, 낮은 레이턴시 필수 var sampleSettings = importer.defaultSampleSettings; // 짧은 클립은 비압축 sampleSettings.loadType = AudioClipLoadType.DecompressOnLoad; sampleSettings.compressionFormat = AudioCompressionFormat.PCM; sampleSettings.sampleRateSetting = AudioSampleRateSetting.OptimizeSampleRate; importer.defaultSampleSettings = sampleSettings; importer.forceToMono = true; // SFX는 모노로 충분 importer.preloadAudioData = true; } private void ConfigureBGM(AudioImporter importer) { // BGM: 스트리밍, 메모리 최소화 var sampleSettings = importer.defaultSampleSettings; sampleSettings.loadType = AudioClipLoadType.Streaming; sampleSettings.compressionFormat = AudioCompressionFormat.Vorbis; sampleSettings.quality = 0.5f; // 50% 품질 (BGM은 괜찮음) importer.defaultSampleSettings = sampleSettings; importer.forceToMono = false; // 스테레오 유지 importer.preloadAudioData = false; // 모바일 별도 설정 (더 낮은 품질) var mobileSettings = sampleSettings; mobileSettings.quality = 0.4f; importer.SetOverrideSampleSettings("Android", mobileSettings); importer.SetOverrideSampleSettings("iPhone", mobileSettings); } private void ConfigureVoice(AudioImporter importer) { // 음성: 압축 인메모리, 명료도 중요 var sampleSettings = importer.defaultSampleSettings; sampleSettings.loadType = AudioClipLoadType.CompressedInMemory; sampleSettings.compressionFormat = AudioCompressionFormat.Vorbis; sampleSettings.quality = 0.7f; // 음성은 품질 중요 importer.defaultSampleSettings = sampleSettings; importer.forceToMono = true; // 음성은 모노 } }

핵심 요약: 로딩 속도 vs 파일 크기

  • 트레이드오프 본질 - I/O 속도 vs CPU 해제 비용, 더 느린 쪽을 최적화
  • 플랫폼별 차이 - NVMe SSD는 비압축, eMMC/HDD는 압축 유리
  • 이중 압축 전략 - 다운로드=LZMA, 런타임=LZ4로 양쪽 최적화
  • LZ4 권장 - 청크 기반 해제로 필요한 것만 로딩
  • GPU 텍스처 포맷 - ASTC/BC7/ETC2로 해제 없이 바로 사용
  • 오디오 분류 - SFX=즉시로드, BGM=스트리밍, Voice=압축인메모리
  • 실측 필수 - 타겟 디바이스에서 직접 벤치마크
